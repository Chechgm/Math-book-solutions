\section{Universal properties determine an object up to unique isomorphism}


\begin{exercise}{1.2.A}
Show that any two initial objects are uniquely isomorphic.
Show that any two final objects are uniquely isomorphic.
\end{exercise}
\begin{proof}
Let $A$ and $B$ be initial.
Then for any $Hom(A, C)$ and $Hom(B, C)$ consist of only one object for all $C \in \cC$.
In particular, $Hom(A, B) = f$ and $Hom(B, A) = g$ constitute an isomorphism between $A$ and $B$, since $g \circ f: $
\end{proof} 

\begin{exercise}{1 (Micro)}
Let $P$ be a poset and consider it as a category. 
What are the initial and final objects in this category?
\end{exercise}
\begin{proof}
The initial objects are the greatest element of the poset;
that is, those elements that are greater than any other elements.
The final objects would be the least elements.
\end{proof} 

\begin{exercise}{2 (Micro)}
Take the category $SET$ with as objects all sets and morphisms all functions, what are the initial and final objects in this category?
\end{exercise}
\begin{proof}
The final objects would be the singletons.
\end{proof}
