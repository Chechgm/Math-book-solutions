\section{5C Lebesgue integration on $R^n$}


\begin{exercise}{3}
Suppose $(X, \cS), (Y, \cT)$ and $(Z, \cU)$ are measurable spaces.
We can define $\cS \otimes \cT \otimes \cU$ to be the smallest $\sigma$-algebra $X \times Y \times Z$ that contains 
\begin{align*}
    \set{A \times B \times C: A \in \cS, B \in \cT, C \in \cU}.
\end{align*}
Prove that if we make the obvious identifications of the products $(X \times Y) \times Z$ and $X \times (Y \times Z)$ with $X \times Y \times Z$, then
\begin{align*}
    \cS \otimes \cT \otimes \cU 
    = (\cS \otimes \cT) \otimes \cU 
    = \cS \otimes (\cT \otimes \cU).
\end{align*}
\end{exercise}
\begin{proof}
We will prove this by double containment.

($\subseteq$)
Let $C \in \cU$ and let $\cE = \set{E \in X \times Y: E \times C \in \cS \otimes \cT \otimes \cU}$.
We want to prove that $\cE$ is a $\sigma$-algebra containing all rectangles of the form $A \times B \times C$ where $A \in \cS$, $B \in \cT$ and $C \in \cU$, so that $\cS \otimes \cT \otimes \cU \subseteq (\cS \otimes \cT) \otimes \cU$.

First, we prove that $\cS \otimes \cT \subseteq \cE$.
For this purpose, suppose $E \in \cS \otimes \cT$.
Since $\cS \otimes \cT \otimes \cU$ contains all sets such that $A \times B \times C$ with $A \in \cS, B \in \cT$ and $C \in \cU$, then for $C \in \cU$, we have that $E \times C \in \cS \otimes \cT \otimes \cU$ because $E$ is an element of a $\sigma$-algebra generated by sets in $\cS$ and $\cT$ so for a fixed $C \in \cU$, $E \times C$ will be constructed in the same way it is constructed in $\cS \otimes \cT$.

We will now prove $\cE$ is a $\sigma$-algebra.
The empty set is a subset of both $X \times Y$ and an element of $\cS \otimes \cT \otimes \cU$ so that $\emptyset \in \cE$.
Now let $E \in \cE$, so that $E \times C \in \cS \otimes \cT \otimes \cU$.
We have that $(E \times C)^C = (E^C \times Z) \cap (X \times Y \times C^C)$.
Both sets on the right belong to $\cS \otimes \cT \otimes \cU$, so their intersection does too.
Finally, let $(E_n)$ be a sequence in $\cE$.
We have that $(\bigcup E_i) \times C = \bigcup (E_i \times C) \in \cS \otimes \cT \otimes \cU$, so that the countable union belongs to $\cE$, as required.

($\supseteq$) To prove the containment in the other direction, we know that $(\cS \otimes \cT) \otimes \cU$ is contained in $\cS \otimes \cT \otimes \cU$ because $\cS \otimes \cT$ is the smallest $\sigma$-algebra containing all the products of measurable sets in $\cS$ and $\cT$ and $(\cS \otimes \cT) \otimes \cU$ is the $\sigma$-algebra containing all the products of measurable sets in $(\cS \otimes \cT)$ and $\cU$, the former, we just concluded, contains all products of measurable sets in $\cS$ and $\cT$ so that $(\cS \otimes \cT) \otimes \cU$ is a $\sigma$-algebra containing all the products of the three $\sigma$-algebras.
\end{proof} 

\begin{exercise}{4}
Show that Lebesgue measure on $\R^n$ is translation invariant.
More precisely, show that if $E \in \cB_n$ and $a \in \R^n$, then $a + E \in \cB_n$ and $\lambda_n (a + E) = \lambda_n(E)$, where $a + E = \set{a + x: x \in E}$.
\end{exercise}
\begin{proof}
We will proceed in a similar way as in 5.41.
Let $\cE = \set{E \in \cB_n: a+E \in \cB_n}$.
Notice that $\cE$ contains every open subset of $\R^n$ (because if $E$ is open, $a + E$ is open).
We will now prove $\cE$ is a $\sigma$-algebra.
$\cE$ is closed under complements because
\begin{align*}
    a + (\R^n \setminus E) &\iff a+x: x \in \R^n \text{ and } x \notin E\\
    &\iff x \in \R^n: x \notin a + E\\
    &\iff \R^n \setminus (a+E).
\end{align*}
To see $\cE$ is closed under countable unions, let $(E_n)$ be a sequence in $\cE$.
We have that $a + \bigcup E_i = \bigcup (a + E_i)$, so that $\bigcup (a + E_i) \in \cE$.

Hence, $\cE$ is a $\sigma$-algebra in $\R^n$ containing the open subsets of $\R^n$, so that $\cB_n \subseteq \cE$, which is the same as $a + \cE \in \cB_n$ for all $E \in \cB_n$.

We will now prove that $\lambda_n(a + E) = \lambda_n(E)$.
We will proceed in two steps:
first we will prove this is the case for products of rectangles in $\cB \times \cB_{n-1}$, and then we will prove it for general sets.
The first step follows by induction.
For the base case, the translation invariance of the outer measure (2.7) tells us that $\lambda_1(a + E) = \lambda_1(E)$.
Now assume the relation holds for $n - 1$, and let $A \in \cB_{n-1}$ and $B \in \cB_1$.
Then,
\begin{align*}
    \lambda_n(a + (A \times B))
    =& \lambda_n((a + A) \times (a + B))\\
    =& \lambda_{n-1}(a + A) \lambda_1(a + B)\\
    =& \lambda_{n-1}(A) \lambda_1(B)\\
    =& \lambda_{n}(A \times B),
\end{align*}
giving us the desired result for $A \times B$.

We will now prove the result for arbitrary sets in $\cB_n$.
Let $m \in \N$ and define $C_m$ to be the open cube in $\R^n$ centered at the origin and with side length $m$.
Let $\cE_m = \set{E \in \cB_n: E \subseteq C_m \text{ and } \lambda_n(a + E) = \lambda_n(E)}$.
From our previous conclusion about $\lambda_n$ on the rectangles and using the fact that any union of measurable rectangles in $\cB_{n-1} \otimes \cB_1$ can be written as a finite union of disjoint measurable rectangles in $\cB_{n-1} \otimes \cB_1$ (5.13.b), we see that finite unions of measurable rectangles contained in $C_m$ are in $\cE_m$.

Following the same strategy, mutatis mutandis, of exercise 7, we can prove $\cE_m$ is closed under countable increasing unions and countable decreasing intersections, making $\cE_m$ a monotone class.
Since the set of finite unions of measurable rectangles in $\cB_n$ is an algebra on $\R^n$ (5.13.a), we can use the Monotone Class Theorem (5.17) to conclude that $\cE_m$ is the $\sigma$-algebra on $C_m$ consisting of Borel subsets of $C_m$.
Thus $\lambda_n(a + E) = \lambda_n(E)$ for all $E \in \cB_n$ such that $E \subseteq C_m$.
Now suppose $E \in \cB_m$.
Then 2.59 implies that
\begin{align*}
    \lambda_n(a + E)
    =& \lim_{m \to \infty} \lambda_n(a + (E \cap C_m))\\
    =& \lim_{m \to \infty} \lambda_n(E \cap C_m)\\
    =& \lambda_n(E),
\end{align*}
completing the proof.
\end{proof} 

\begin{exercise}{5}
Suppose $f: \R^n \to \R$ is $\cB_n$ measurable and $t \in \R \setminus \set{0}$.
Define $f_t: \R^n \to \R$ by $f_t(x) = f(tx)$.
\begin{enumerate}
    \item Prove that $f_t$ is $\cB_n$-measurable.
    \item Prove that if $\int_\R^n f d\lambda_n$ is defined, then
    \begin{align*}
        \int_\R^n f_t d\lambda_n = \frac{1}{\absoluteValue{t}^n} \int_\R^n f d\lambda_n.
    \end{align*}
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item Let $B$ be a Borel set. 
    We have $f_t^{-1}(B) = f^{-1}(B/t)$, since $f$ is $\cB_n$-measurable, and $B/t$ is Borel (2.B.8) then $f_t^{-1}(B) \in \cB_n$, so that $f_t$ is $\cB_n$-measurable.
    \item We will prove this using the bootstrap method used in the exercises of chapter 3, for simplicity, we will assume $t > 0$.
    For that purpose, we begin by proving the statement for $f = \chi_E$, for $E \in \cB_n$.
    Let $\chi_{E, t}(x) = \chi_E(tx)$.
    We have
    \begin{align*}
        \int_{\R^n} f_t d\lambda_n
        =& \int_{\R^n} \chi_{E, t} d\lambda_n\\
        =& \lambda_n(tE)\\
        =& t^n \lambda_n(E)\\
        =& t^n \int_{\R^n} f d\lambda_n,
    \end{align*}
    as required.

    Now let $f$ be a simple function, so that $f = \sum_k^K a_k \chi_{E_k}$ for disjoint measurable sets $E_k$.
    We have
    \begin{align*}
        \int_{\R^n} f_t d\lambda_n
        =& \int_{\R^n} (\sum_k^K \chi_{t, E_k}) d\lambda_n\\
        =& \sum_k^K a_k \lambda_n(t E_k)\\
        =& \sum_k^K a_k t^n \lambda_n(E_k)\\
        =& t^n \sum_k^K a_k \lambda_n(E_k)\\
        =& t^n \int_{\R^n} f d\lambda_n,
    \end{align*}
    as required.

    Now consider a positive function $f$ and let $0 \leq f^1 \leq f^2 \leq \dots$ be an increasing sequence of simple functions converging to $f$ (we use a upper index because we have reserved the lower index for $t$).
    The existence of such sequence is guaranteed by 2.89.
    Furthermore, if $f^k \to f$, then $f_t^k \to f_t$ (consider $\absoluteValue{f_t^k(x) - f_t(x)} = \absoluteValue{f^k(tx) - f(tx)}$, and find the $K$ that makes the right hand side less than $\epsilon$ for all $k > K$).
    By the Monotone Convergence Theorem (3.11) and the algebraic properties of limits, we have
    \begin{align*}
        \int_{\R^n} f_t d\lambda_n
        = \lim_{k \to \infty} \int_{\R^n} f_t^k d\lambda_n
        = \lim_{k \to \infty} t^n \int_{\R^n} f^k d\lambda_n
        = t^n \int_{\R^n} f d\lambda_n,
    \end{align*}
    as required.

    For a real valued function, we can apply the same strategy, mutatis mutandis to $f^+$ and $f^-$ so that by the definition of the integral on real valued functions, we get the desired result.
\end{enumerate}
\end{proof} 

\begin{exercise}{6}
Suppose $\lambda$ denotes Lebesgue measure on $(\R, \cL)$, where $\cL$ is the $\sigma$-algebra of Lebesgue measurable subsets of $\R$.
Show that there exists subsets $E$ and $F$ of $\R^2$ such that 
\begin{itemize}
    \item $F \in \cL \otimes \cL$ and $(\lambda \times \lambda)(F) = 0$;
    \item $E \subset F$ but $E \notin \cL \otimes \cL$.
\end{itemize}
[The measure space $(\R, \cL, \lambda)$ has the property that every subset of a measurable set with measure 0 is measurable.
This exercise asks you to show that the measure space $(\R^2, \cL \otimes \cL, \lambda \times \lambda)$ does not have this property.]
\end{exercise}
\begin{proof}
Consider the set $F = \set{0} \times [-1,1]$.
We have that $(\lambda \times \lambda)(F) = 0 \cdot 2 = 0$.
Moreover, consider $E = \set{0} \times V$, where $V$ is the Vitali set.
We have $E \subset F$ but $E$ is not in $\cL \otimes \cL$ because $V$ is not Lebesgue measurable.
\end{proof} 

\begin{exercise}{7}
Suppose $m \in \N$.
Verify that the collection of sets $\cE_m$ that appears in the proof of 5.41 is a monotone class.
\end{exercise}
\begin{proof}
Let $C_m$ be the open cube in $\R^n$ centered at 0 with side length $m$.
To see$\cE_m = \set{E \in \cB_n: E \subseteq C_m \text{ and } \lambda_n(tE) = t^n \lambda_n(E)}$, is a monotone class, first let $E_1 \subseteq E_2 \subseteq \dots$ be an increasing sequence in $\cE_m$.
By definition, we have that each $E_i \subseteq C_m$, so that $\bigcup E_i \subseteq C_m$.
Furthermore $\lambda_n(tE_i) = t^n \lambda_n(E_i)$. 
We have 
\begin{align*}
    \lambda_n(t \bigcup E_i)
    = \lambda_n(\bigcup t E_i)
    = \lim_{k \to \infty} \lambda_n(t E_k)
    = \lim_{k \to \infty} t^n \lambda_n(E_k)
    = t^n \lambda_n(\bigcup E_i),
\end{align*}
where we have changed the limits for the countable union using 2.59.
Putting everything together, $\bigcup E_i \in \cE_m$

Now let $E_1 \supseteq E_2 \supseteq \dots$ be a decreasing sequence in $\cE_m$.
Since $E_1 \subseteq C_m$, then $\bigcap E_i \subseteq C_m$, but furthermore $\lambda_n(E_1) < \infty$, so that
\begin{align*}
    \lambda_n(t \bigcap E_i)
    = \lambda_n(\bigcap t E_i)
    = \lim_{k \to \infty} \lambda_n(t E_k)
    = \lim_{k \to \infty} t^n \lambda_n(E_k)
    = t^n \lambda_n(\bigcap E_i),
\end{align*}
so that $\bigcap E_i \in \cE_m$, making $\cE_m$ a monotone class.
\end{proof} 

\begin{exercise}{12}
\begin{enumerate}
    \item Prove that $\lim_{n \to \infty} \lambda_n(\cB_n) = 0$.
    \item Find the value of $n$ that maximises $\lambda_n(\cB_n)$.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item Using the result in exercise 13, for even $n$, we have $\lambda_n(\cB_n) = \frac{\pi^{n/2}}{(n/2)!}$ for $n/2 > \pi^{1/2}$, as $n$ gets larger, the numerator is multiplied by a smaller number $\pi^{1/2}$ than the denominator $n/2$, and thus $\lambda_n(\cB_n) \to 0$.
    We can analyse the case for odd $n$ mutatis mutandis.
    \item The table on page 142 hints at a quickly decelerating volume of the $n$-ball for $n$ up to 5.
    The most direct approach to find the $n$ that maximises the volume is by computing the volume for the subsequent $n$ and seeing where it starts decreasing.
    We have that $\lambda_6(\cB_6) = 5.16$, which is already less than $\lambda_5(\cB_5) = 5.26$, so that the $n$ that maximises the volume of the ball is 5.
\end{enumerate}
\end{proof} 

\begin{exercise}{13}
For readers familiar with the Gamma function $\Gamma:$
Prove that 
\begin{align*}
    \lambda_n(\cB_n) = \frac{\pi^{n/2}}{\Gamma(\frac{n}{2} + 1)}
\end{align*}
for every positive integer $n$.
\end{exercise}
\begin{proof}
If $n$ is even, 5.44 tells us that 
\begin{align*}
    \lambda_n = \frac{\pi^{n/2}}{(n/2)!} = \frac{\pi^{n/2}}{\Gamma (n/2 + 1)},
\end{align*}
since $\Gamma(n) = (n-1)!$ for integers.

Now suppose $n$ is odd.
We have that $n/2$ is a half integer (an integer of the form $n = m + 1/2$, for some integer $m$), since $n-1 = 2m$ for an integer $m$, and so $n/2 = (n-1+2)/2 = 2m/2 + 1/2 = m + 1/2$, which also in turn implies that $(n/2) + 1$ is a half integer.
Knowing that $n$ is a half integer, we can use the following known evaluation of the Gamma function 
\begin{align*}
    \Gamma(\frac{n}{2})
    = \frac{\pi^{1/2}(n-2)!!}{2^{(n-1)/2}}
\end{align*}
where $n!! = n \cdot n-2 \cdots 5 \cdot 3 \cdot 1$.
We then have
\begin{align*}
    \frac{\pi^{n/2}}{\Gamma(\frac{n}{2}+1)}
    = \frac{\pi^{n/2}}{\Gamma(\frac{n + 1}{2})}
    = \frac{\pi^{n/2}}{\frac{\pi^{1/2}(n-2+2)!!}{2^{(n+2-1)/2}}}
    = \frac{\pi^{(n-1)/2}2^{(n+1)/2}}{n!!},
\end{align*}
which is precisely the expression derived in 5.44.
\end{proof}
