\section{5B Iterated integrals}


\begin{exercise}{1}
\begin{enumerate}
    \item Let $\lambda$ denote Lebesgue measure on $[0,1]$.
    Show that 
    \begin{align*}
        \int_{[0,1]} \int_{[0,1]} \frac{x^2 - y^2}{(x^2 + y^2)^2} d\lambda(y) d\lambda(x)
            = \frac{\pi}{4}
    \end{align*}
    and
    \begin{align*}
        \int_{[0,1]} \int_{[0,1]} \frac{x^2 - y^2}{(x^2 + y^2)^2} d\lambda(x) d\lambda(y)
            = -\frac{\pi}{4}
    \end{align*}
    \item Explain why this violates neither Tonelli's Theorem nor Fubini's Theorem
\end{enumerate}
\end{exercise}
\begin{proof}
We have the following:
\begin{align*}
    \int_{[0,1]} \int_{[0,1]} \frac{x^2 - y^2}{(x^2 + y^2)^2} d\lambda(y) d\lambda(x)
    =& \int_{[0,1]}  \sqrBrackets{\frac{y}{x^2 + y^2}}^1_0  d\lambda(x)\\
    =& \int_{[0,1]} \frac{1}{x^2 + 1} d\lambda(x)\\
    =& \arctan(x)\lvert^1_0\\
    =& \frac{\pi}{4},
\end{align*}
and
\begin{align*}
    \int_{[0,1]} \int_{[0,1]} \frac{x^2 - y^2}{(x^2 + y^2)^2} d\lambda(x) d\lambda(y)
    =& \int_{[0,1]}  -\sqrBrackets{\frac{x}{x^2 + y^2}}^1_0  d\lambda(y)\\
    =& \int_{[0,1]} -\frac{1}{y^2 + 1} d\lambda(y)\\
    =& -\arctan(y)\lvert^1_0\\
    =& -\frac{\pi}{4},
\end{align*}
as required.

This doesn't violate Fubini's Theorem, because $\int_{X \times Y} \absoluteValue{f(x)} d(\mu \times \nu) = \infty$, and doesn't violate Tonelli's Theorem because the function takes on negative values.
\end{proof} 

\begin{exercise}{2}
\begin{enumerate}
    \item Give an example of a doubly indexed collection $\set{x_{m, n}: m, n \in \N}$ of real numbers such that
    \begin{align*}
        \sum_{m=1}^\infty \sum_{n=1}^\infty x_{m,n} = 0
    \end{align*}
    and
    \begin{align*}
        \sum_{n=1}^\infty \sum_{m=1}^\infty x_{m,n} = \infty
    \end{align*}
    \item Explain why this violates neither Tonelli's Theorem nor Fubini's Theorem.
\end{enumerate}
\end{exercise}
\begin{proof}
Consider the doubly indexed sequence $(a_{n,m})$ given by $a_{n,m} = 1/n$ if $m=1$, $a_{n,m} = -1/n$ if $m=2$ and $0$ if $m > 2$.
We have that $\sum_m \sum_n a_{n,m} = \infty$, whereas $\sum_n \sum_m a_{n,m} = 0$.

This sequence does not violate Fubini's Theorem, because $\int_\N \absoluteValue{a_{n,m}} d\mu(n) = \sum_n \absoluteValue{a_{n,m}} = \infty$ (and Fubini's Theorem requires it to be finite), and does not violate Tonelli's Theorem because $a_{n,m}$, understood as a function on $\N \times \N$ has negative values.
\end{proof} 

\begin{exercise}{3}
Suppose $(X, \cS)$ is a measurable space and $f: X \to [0, \infty]$ is a function.
Let $\cB$ denote the $\sigma$-algebra of Borel subsets of $(0, \infty)$.
Prove that $U_f \in \cS \otimes \cB$ if and only if $f$ is an $\cS$-measurable function.
\end{exercise}
\begin{proof}
($\Rightarrow$)
We have
\begin{align*}
    f^{-1}((a, \infty))
    =& \set{x \in X: a < f(x)}\\
    =& [ \set{(x, t) \in X \times (0,\infty): 0 < t < f(x)} ]_a\\
    =& [U_f]_a,
\end{align*}
which we know is an $\cS$-measurable set because $U_f \in \cS \otimes \cB$ by assumption and 5.6.

($\Leftarrow$)
This is the content of the first part of Proposition 5.35, noticing that to prove that part we used only the fact that $f$ is an $\cS$-measurable function.
\end{proof}
