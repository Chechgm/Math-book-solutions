\section{6C Normed Vector Spaces}


\begin{exercise}{14}
Suppose $U$ is a subspace of a normed vector space $V$.
Suppose also that $W$ is a Banach space and $S: U \to W$ is a bounded linear map.
\begin{enumerate}
    \item Prove that there exists a unique continuous function $T: \bar{U} \to W$ such that $T|_U = S$.
    \item Prove that the function $T$ in the previous point is a bounded linear map from $\bar{U}$ to $W$ and $\norm{T} = \norm{S}$.
    \item Give an example to show that in the first point can fail if the assumption that $W$ is a Banach space is replaced by the assumption that $W$ is a normed vector space.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item Let $(x_n)$ be a sequence in $U$ with $x_n \to x$.
    We have that 
    \begin{align*}
        \norm{S(x_n) - S(x_m)} 
        \leq \norm{S(x_n - x_m)} 
        \leq \norm{S}\norm{x_n - x_m},
    \end{align*}
    so that $(S(x_n))$ is Cauchy, and since $W$ is a Banach space, $S(x_n) \to y \in W$.
    Let $Tx = y = \lim S(x_n)$.

    We have that $T$ is well defined because of the continuity of $S$, so that if $x_n \to x$ and $x_n' \to x$, then 
    \begin{align*}
        \lim S(x_n) 
        = S(\lim x_n) 
        = S(x) 
        = S(\lim x_n') 
        = \lim S(x_n').
    \end{align*}

    Furthermore, $T|_U = S$ by definition, and $T$ is a linear map by the algebraic properties of limits.
    The continuity of $T$ will follow by the fact that $T$ bounded (6.48) which we prove in the next numeral.
    
    We will use the continuity of $T$ to prove it must be unique.
    To see this, suppose there is another linear map $T'$ such that $T'|_U = S$, we then have $\norm{T(x_n) - T'(x_n)} = \norm{S(x_n) - S(x_n)} = 0$, so that $T = T'$.
    
    \item We have $\norm{Tx} = \norm{S(x_n)} \leq \norm{S}\norm{x_n} = \norm{S}\norm{x}$.
    Hence, $\norm{Tx} \leq \norm{S}\norm{x}$, meaning $T$ is bounded. 
    Of course since $T$ is an extension of $S$, $\norm{T} \geq \norm{S}$ giving us that $\norm{T} = \norm{S}$.
    
    \item Let $W$ be the subspace of $C[0,1]$ of polynomials.
    Consider the identity, which is a linear map linear map $I: W \to W$.
    Now let $f(x) = e^x \in C[0,1]$, and because $W$ is dense in $C[0,1]$, we know there is a sequence of polynomials $(f_n)$, such that $f_n(x) \to e^x$.
    This means that $(f_n)$ is a Cauchy sequence in $W$ which converges to a point outside of $W$.
    But we cannot extend $I$ to a continuous linear map $I': C[0,1] \to W$, we would then have that $f_n \to e^x$ and $I'(f_n) = f_n \to I'(e^x)$, meaning that $(f_n)$ actually converges in $W$, which we have argued is not true.
\end{enumerate}
\end{proof} 

\begin{exercise}{15}
For readers familiar with the quotient of a vector space and a subspace:
Suppose $V$ is a normed vector space and $U$ is a subspace of $V$.
Define $\norm{\cdot}$ on $\quot{V}{U}$ by $\norm{f+ U} = \inf\set{\norm{f + g}: g \in U}$.
\begin{enumerate}
    \item Prove that $\norm{\cdot}$ is a norm on $\quot{V}{U}$ if and only if $U$ is a closed subspace of $V$.
    \item Prove that if $V$ is a Banach space and $U$ is a closed subspace of $V$, then $\quot{V}{U}$ (with the norm defined above) is a Banach space.
    \item Prove that if $U$ is a Banach space (with the norm it inherits from $V$) and $\quot{V}{U}$ is a Banach space (with the norm defined above), then $V$ is a Banach space.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item ($\Rightarrow$)
    By contrapositive, suppose $U$ is not a closed subspace of $V$.
    Then there exists a sequence $(f_n)$ in $U$ such that $f_n \to f$ but $f \notin U$.
    That is, $\norm{f + U} = \inf{\norm{f + g}: g \in U} = 0$ (for any $\epsilon > 0$, consider $g = -f_N \in U$ for the required $N$), but $f + U$ is not the 0 element of $\quot{V}{U}$, so that $\norm{\cdot}$ is not a norm on $\quot{V}{U}$.

    ($\Leftarrow$)
    Suppose $U$ is a closed subspace of $V$.

    (Positive definite) 
    Let $f + U \in \quot{V}{U}$.
    Then $\norm{f + U} = \inf\set{\norm{f + g}: g \in U}$, but such infimum is 0 if and only if $f \in U$, since $\norm{f - g} > \epsilon$ for all $g \in U$.
    If this were not the case, $U$ wouldn't be closed.

    (Homogeneity)
    Let $\alpha \in \bF$.
    We have 
    \begin{align*}
        \norm{\alpha(f + U)} 
        =& \inf\set{\norm{\alpha(f + g)}: g \in U}\\
        =& \inf\set{\absoluteValue{\alpha}\norm{f + g}: g \in U}\\
        =& \absoluteValue{\alpha} \inf\set{\norm{f + g}: g \in U}\\
        =& \absoluteValue{\alpha}\norm{f + U},
    \end{align*}
    as required.

    (Triangle inequality)
    Let $f' + U \in \quot{V}{U}$.
    We have
    \begin{align*}
        \norm{f + U + f' + U}
        =& \norm{f + f' + U}\\
        =& \inf\set{\norm{f + f' + g}: g \in U}\\
        =& \inf\set{\norm{f + f' + g + g'}: g, g' \in U}\\
        \leq& \inf\set{\norm{f + g} + \norm{f' + g'}: g, g' \in U}\\
        \leq& \inf\set{\norm{f + g}: g \in U} + \inf\set{\norm{f' + g'}: g' \in U}\\
        =& \norm{f + U} + \norm{f' + U},
    \end{align*}
    completing the proof.

    \item We will prove this via 6.41;
    that is, we will prove that if a sequence $(f_n + U)$ converges absolutely, $\sum \norm{f_n + U)} < \infty$ means the sequence converges $\sum (f_n + U)$, then $\quot{V}{U}$ is a Banach space.
    Thus, suppose $\sum (f_n + U) < \infty$.
    We can choose a sequence with elements $F_n = f_n + g_n \in  (f_n + U)$ such that $\norm{F_n} \leq 2\norm{f_n + U}$.
    We then have that $\sum \norm{F_n} \leq 2\sum \norm{f_n + U} < \infty$ so that $\sum F_n \to F \in V$ (because V is Banach).

    Now let $\pi: V \to \quot{V}{U}$ be the quotient map which is continuous (because $\norm{\pi(f)} = \inf_{g \in U} \norm{f + g} \leq \norm{f}$, so that it is bounded and hence uniformly continuous), we then have that 
    \begin{align*}
        \sum (f_n + U) 
        = \sum \pi(F_n) 
        = \pi(\sum F_n) 
        = \pi(F) 
        = F + U,
    \end{align*}
    where we exchanged summation and $\pi$ because of the continuity of $\pi$.
    Thus, $\sum (f_n + U)$ converges to $F + U$ and by 6.41 $\quot{V}{U}$ is Banach.


    (Alternative proof)
    Let $(f_n + U)$ be Cauchy.
    Thus, we can find a subsequence, $(f_{n_k} + U)$ with $\norm{(f_{n_{k+1}} + U) - (f_{n_k} + U)} < 1/2^k$ for all $k \geq 1$, for all $k \geq 1$.
    By the definition of the infimum, we can find $g_k \in U$ with $\norm{f_{n_{k+1}} - f_{n_k} + g_k} < 1/2^k$.

    Let $x_1 = f_{n_1}$ and more generally $x_{k+1} = f_{n_{k+1}} + \sum_{i=1}^k g_i$.
    We have that $x_k + U = f_{n_k} + U$ because $U$ is closed under addition, meaning that $\sum_{i=1}^k g_i \in U$.
    Thus, $x_k + U$ is a valid representative of $f_{n_k} + U$.

    We have $\norm{x_{k+1} - x_k} = \norm{f_{n_{k+1}} - f_{n_k} + g_k} < 1/2^k$, meaning that $(x_k)$ is Cauchy.
    Thus $x_k \to x$ because $V$ is Banach.

    Now we know that the quotient map $\pi: V \to \quot{V}{U}$ is continuous (since $\norm{\pi(f)} = \inf_{g \in U} \norm{f + g} \leq \norm{f}$), then $(f_{n_k} + U) = (x_k + U) \to (x + U)$.
    That is, $(f_n + U)$ is a Cauchy sequence with a convergent subsequence, which means that $(f_n + U)$ is itself convergent, with limit $x + U$.
    
    \item Suppose $(f_n)$ is a Cauchy sequence in $V$.
    We saw in the previous numeral that $\pi: V \to \quot{V}{U}$, the quotient function, is uniformly continuous, so that $(\pi(f_n)) = (f_n + U)$ is Cauchy in $\quot{V}{U}$.
    By assumption $(f_n + U)$ is also convergent: $(f_n + U) \to (f + U)$ (where we can choose $f \in V$ since $\pi$ is surjective).
    That is, for all $\epsilon > 0$, we can find an $N \in \N$ such that whenever $n > N$, it holds that $\norm{(f_n + U) - (f + U)} = \norm{f_n - f + U} = \inf_{g \ \in U}\norm{f_n - f + g} < \epsilon$.

    We can now choose a subsequence of $(f_{n_k})$ such that $\norm{f_{n_k} - f + U} < 1/2^k$, so that there exists $g_k \in U$ with $\norm{f_{n_k} - f + g_k} < 1/2^k$ for all $k$.
    Note that $(g_k)$ is Cauchy:
    \begin{align*}
        \norm{g_k - g_j}
        =& \norm{f_{n_k} - f + g_k - f_{n_j} + f - g_j - f_{n_k} + f_{n_j}}\\
        \leq& \norm{f_{n_k} - f + g_k} + \norm{f_{n_j} - f + g_j} + \norm{f_{n_k} - f_{n_j}}\\
        <& 1/2^k + 1/2^j + \epsilon,
    \end{align*}
    (where the $\epsilon$ follows by choosing a suitable $N \in \N$ and $n_k, n_j > N$).
    The right hand side of the inequality converges to 0 as $k, j \to \infty$.
    Since $U$ is Banach, $g_k \to g \in U$.
    We have that $\lim_{k \to \infty} f_{n_k} = \lim_{k \to \infty} (f_{n_k} - f + g_k) + (f - g_k) = 0 + f - g_k \in V$, which completes the proof, as any Cauchy sequence with a convergent subsequence converges itself.
\end{enumerate}
\end{proof} 

\begin{exercise}{16}
Suppose $V$ and $W$ are normed vector spaces with $V \neq \set{0}$ and $T: V \to W$ is a linear map.
\begin{enumerate}
    \item Show that $\norm{T} = \sup \set{\norm{Tf}: f \in V \text{ and } \norm{f} < 1}$.
    \item Show that $\norm{T} = \sup \set{\norm{Tf}: f \in V \text{ and } \norm{f} = 1}$.
    \item Show that $\norm{T} = \inf \set{c \in [0, \infty): \norm{Tf} \leq c\norm{f} \text{ for all } f \in V}$.
    \item Show that $\norm{T} = \sup \set{\frac{\norm{Tf}}{\norm{f}}: f \in V \text{ and } f \neq 0}$.
\end{enumerate}
\end{exercise}
\begin{proof}
Before proving these formulas, notice that the definition of the norm of $T$ given by Axler is the following: 
\begin{align*}
    \norm{T} 
    = \sup \set{\norm{Tf}: f \in V \text{ and } \norm{f} \leq 1}
    = \sup_{f \in V, \norm{f} \leq 1} \norm{Tf}.
\end{align*}

\begin{enumerate}
    \item Let $A = \set{\norm{Tf}: f \in V \text{ and } \norm{f} \leq 1}$ and $B = \set{\norm{Tf}: f \in V \text{ and } \norm{f} < 1}$.
    We want to prove $\sup A = \sup B$.
    We will prove this via two inequalities.
    We get directly that $\sup A \geq \sup B$, since $B \subseteq A$.
    
    To see the other direction holds, let $\norm{T} = \sup A$, so that by the definition of the supremum, for all $\epsilon > 0$, there exists an $f \in V$ with $\norm{T} - \epsilon \leq \norm{Tf}$.
    If $\norm{f} < 1$, we are done, since then $f \in B$.
    Thus suppose $\norm{f} = 1$ and let $k_n = 1- 1/n$ and $f_n = k_n f$.
    For all $n$, we have that $f_n \in B$ and moreover $f_n \to f$.
    We have $\lim \norm{Tf_n} = \lim (1 - 1/n)\norm{Tf} = \norm{Tf}$, and since $\sup B \geq \norm{Tf_n}$, we have that $\sup B \geq \norm{Tf} > \norm{T} - \epsilon$.
    Since this holds for all $\epsilon$, we have that $\sup B \geq \norm{T}$, completing the proof.
    
    \item Let $A = \set{\norm{Tf}: f \in V \text{ and } \norm{f} \leq 1}$ and $B = \set{\norm{Tf}: f \in V \text{ and } \norm{f} = 1}$.
    We want to prove $\sup A = \sup B$.
    We will prove this via two inequalities.
    We get directly that $\sup A \geq \sup B$, since $B \subseteq A$.
    
    To see the other inequality holds, let $f \in V$ be such that $\norm{f} \leq 1$.
    We have $\norm{Tf} \leq \norm{Tf}/\norm{f} = \norm{Tg}$, for some $g$ with $\norm{g} = 1$.
    Taking supremum on both sides of the inequality gives us the desired result.
    
    \item Let $A = \set{\norm{Tf}: f \in V \text{ and } \norm{f} = 1}$ and $B = \set{c \in [0, \infty): \norm{Tf}/\norm{f} \leq c \text{ for all } f \in V, f \neq 0}$.
    As before, we will prove the statement using a double inequality.

    First, if $\norm{T} = \sup A$, then for any $f \in V$, with $f \neq 0$, we have $\norm{T(f/\norm{f})} = \norm{Tf}/\norm{f} = \norm{Tg} \leq \norm{T}$, for some $g$ with $\norm{g} = 1$ so that $\inf B \leq \sup A$.

    Take any $c \in B$, so that for any $f \in V$ with $f \neq 0$, $\norm{Tf} / \norm{f} \leq c$, restricting this to the vectors with $\norm{f} = 1$, we have $\norm{Tf} \leq c$.
    Now taking the supremum on the left hand side tells us that $\sup A \leq c$, so that $\sup A$ is a lower bound of $B$, meaning that $\sup A \leq \sup B$.
    
    \item From the second numeral, we know $\norm{T} = \sup \set{\norm{Tf}: f \in V \text{ and } \norm{f} = 1}$.
    For all $f \neq 0 $, we have $\norm{Tf}/\norm{f} = \norm{Tg}$, where $\norm{g} = 1$, so that taking the supremum on both sides of the inequality, gives us the desired result.
\end{enumerate}
\end{proof}
