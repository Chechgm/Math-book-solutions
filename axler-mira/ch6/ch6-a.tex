The exercises here were assigned by my mentor because the exercises in the book are too basic.

\begin{exercise}{(Extra) 1}
Let $(X, \cS, m)$ be a finite measure space. 
For $E, F \in \cS$, let $E \triangle F$ be the symmetric difference. 
Let the distance $d(E,F)$ be defined by $m(E \triangle F)$. 
\begin{enumerate}
    \item Show that $d$ is a pseudometric.
    \item If we define $E \sim F$ if and only if $d(E,F) = 0$, show that this is an equivalence relation.
    \item Define $Y = \quot{\cS}{\sim}$ the set of equivalence classes of $\sim$. 
    Show that d descends in a natural way to a metric on $Y$.
    \item Show that $Y$ is a complete metric space.
    \item Show that $m:\cS \to \R$ descends to a function $Y \to R$ that is continuous.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item For $E \in \cS$, we have that $E \triangle E = \emptyset$, so that $d(E, E) = m(E \triangle E) = m(\emptyset) = 0$.
    The symmetric difference is symmetric by definition, so that for $E, F \in \cS$, $d(E, F) = m(E \triangle F) = m(F \triangle E) = d(F, E)$.
    Finally, let $D \in \cS$.
    We will prove $E \triangle F \subseteq E \triangle D \cup D \triangle F$.
    We have that
    \begin{align*}
        x \in E \triangle F \iff x \in E \cup F \text{ and } x \notin E \cap F.
    \end{align*}
    On the other hand,
    \begin{align*}
        x \in (E \triangle D) \cup (D \triangle F) 
        \iff& (x \in E \cup D \text{ and } x \notin E \cap D) \text{ or }\\
        &(x \in D \cup F \text{ and } x \notin D \cap F).
    \end{align*}
    Suppose then that $x \in E \triangle F$ and without loss of generality, suppose $x \in E$.
    If $x \in D$, then  $x \in (E \triangle D) \cup (D \triangle F)$, by the second condition above (since $x \notin F)$.
    The case for $x \in F$ and $x \notin E$ is completely symmetric, so that $E \triangle F \subseteq (E \triangle D) \cup (D \triangle F)$. 
    
    We have that 
    \begin{align*}
        d(E, F) 
        =& m(E \triangle F) \\
        \leq& m((E \triangle D) \cup (D \triangle F))\\
        \leq& m(E \triangle D) + m(D \triangle F)\\
        =& d(E, D) + d(D, F),
    \end{align*}
    where the first inequality follows by the order preserving properties of measures (2.57) and the second inequality follows by the countable subadditivity of measures (2.58).

    This proves $d$ is a pseudometric.
    Notice that $d$ is not a metric.
    Consider $(X, \cS, m)$ with $X = \set{1,2, 3}$, $\cS$ the discrete $\sigma$-algebra, and $m$ as follows $m(X) = 1$ and $m(A) = 0$ for any $A \subseteq X$ with $A \neq X$.
    Then $d(\set{1}, \set{2}) = m(\set{1, 2}) = 0$, but $\set{1} \neq \set{2}$.
    \item For $E \in \cS$, $E \triangle E = \emptyset$, so that $d(E, E) = 0$ and $\sim$ is reflexive. 
    The symmetric difference is symmetric so that $E \triangle F = F \triangle E$ giving us that $d(E, F) = d(F, E)$, regardless of the value of $d(E, F)$, so it certainly holds for $d(E, F) = 0$.
    Finally, let $D \in \cS$, and suppose $d(E, F) = 0$ and $d(F, D) = 0$.
    This implies either that $E$ and $F$ are the same set or that they have no elements in common and likewise for $F$ and $D$.
    In both cases, this implies that $E$ and $D$ are either the same, or that they have no elements in common, which implies $d(E, D) = 0$ so that $\sim$ is transitive, finishing the proof that $\sim$ is an equivalence relation.
    \item We only need to prove that $d$ defined on $\quot{Y}{\sim}$ has the property that $E, F \in Y = \quot{\cS}{\sim}$ with $d(E, F) = 0$ if and only if $E = F$.
    To prove this, notice that $d(E, F) = 0$ if and only if $E \sim F$ by the definition of the quotient space, so that $E$ and $F$ belong to the same equivalence class;
    that is, $E = F$.
    \item Let $(E_n)$ be Cauchy in $Y$.
    Thus, for all $\epsilon >0$, there exists an $N \in \N$, such that whenever $n,m > N$, it holds that $d(E_n, E_m) < \epsilon$
\end{enumerate}
\end{proof} 

\begin{exercise}{(Extra) 2}
Let $(X,d)$ be any metric space. 
Let the Borel algebra of $(X,d)$ be defined as the $\sigma$-algebra generated by the open sets.
\begin{enumerate}
    \item Show that this extends the notion of Borel $\sigma$-algebra you've already seen.
    \item Show that all closed sets are measurable.
    \item If $f$ is a continuous function between metric spaces, show that it is measurable
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item This extends the notion of Borel $\sigma$-algebra we have seen by considering sets different from $\R$ and metrics different from the usual metric on $\R$.
    \item Let $\cS$ be the $\sigma$-algebra generated by the open sets in $(X, d)$.
    For any closed set $F \subseteq X$, we have that $F^C$ is open, since every open set is in $\cS$ and every complement of a measurable set is measurable, then $F$ is measurable.
    \item We have that the preimage of an open set of a continuous function is itself open.
    Thus the open set is a Borel set, and its preimage is also an open set (and by definition, measurable), so that the function is measurable.
\end{enumerate}
\end{proof}
