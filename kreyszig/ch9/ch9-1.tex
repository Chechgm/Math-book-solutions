\begin{exercise}{1}
It was mentioned in the text that for all self-adjoint linear operator $T$ the inner product $\brackets{Tx, x}$ is real.
What does this imply for matrices?
What familiar theorem on matrices does Theorem 9.1-1 include as a special case?
\end{exercise}
\begin{proof}
Let $A$ be the matrix representing the operator $T$.
If $\brackets{Ax, x}$ is real, then $\bar{x}^T A x$ is real, furthermore, since $T$ is self-adjoint, $A$ is Hermitian (the name for self-adjoint in matrices) so that its eigenvalues are all real and the corresponding eigenvectors are orthogonal.
\end{proof} 

\begin{exercise}{2}
If in the finite dimensional case, a self-adjoint linear operator $T$ is represented by a diagonal matrix, show that the matrix must be real.
What is the spectrum of $T$?
\end{exercise}
\begin{proof}
Let $x, y \in X$, where $X$ is finite dimensional.
Thus, the inner product in $X$ is equivalent to the dot product:
$\brackets{x, y} = \sum x_i y_i$. 
Let $T$ be a self-adjoint linear operator whose matrix is diagonal.
We have $\brackets{Tx, y} = \sum \lambda_i x_i y_i$ and $\brackets{x, Ty} = \sum \overline{\lambda_i} x_i y_i$.
For these expressions to be equal, we compare the coefficients of the products $x_i y_i$ for each $i$ (since this has to hold for all $x$ and $y$ and we can simply choose elements of the canonical basis individually).
We need $\lambda_i = \overline{\lambda_i}$, so that the elements of the diagonal are real.
The chosen notation for the elements of the diagonal was suggestive: 
the spectrum of $T$ are the diagonal elements of the matrix.
\end{proof} 

\begin{exercise}{4}
Illustrate Theorem 9.1-2 by an operator whose spectrum consists of a single given value $\lambda_0$.
What is the largest $c$ in this case?
\end{exercise}
\begin{proof}
In a finite dimensional space of any fixed dimension, the identity matrix (which is self-adjoint) has a single eigenvalue, namely 1.
In this case, the largest $c$ is 0, since $I_1 = 0$, so that $\norm{I_1 x} = 0 \geq c \norm{x}$ holds for all $x$ only when $c = 0$. 
\end{proof} 

\begin{exercise}{5}
Let $T: H \to H$ and $W: H \to H$ be bounded linear operators on a complex Hilbert space $H$.
If $T$ is self-adjoint, show that $S = W^\ast T W$ is self-adjoint.
\end{exercise}
\begin{proof}
We have 
\begin{align*}
    \brackets{W^\ast T W x, x}
    = \brackets{T W x, W x}
    = \brackets{W x, T W x}
    = \brackets{x, W^\ast T W x},
\end{align*}
where in the second equality we used that $T$ is self-adjoint, thus proving the desired result.
\end{proof} 

\begin{exercise}{6}
Let $T: \ell^2 \to \ell^2$ be defined by $(x_1, x_2, \dots) \mapsto (0, 0, x_1, x_2, \dots)$.
Is $T$ bounded?
Self-adjoint?
Find $S: \ell^2 \to \ell^2$ such that $T = S^2$.
\end{exercise}
\begin{proof}
$T$ is bounded, since $\norm{Tx} = (\sum \absoluteValue{x_i})^{1/2} = \norm{x}$ for all $x$.
Furthermore, for $y \in \ell^2$, we have $\brackets{Tx, y} = \sum x_n y_{n+2} \neq \sum x_{n+2} y_n = \brackets{x, Ty}$, so that $T$ is not self-adjoint.
The operator $S: \ell^2 \to \ell^2$ given by $S = (0, x_1, x_2, \dots)$ is such that $S^2 = T$.
\end{proof} 

\begin{exercise}{7}
Let $T: \ell^2 \to \ell^2$ be defined by $y = (y_n) = T x$, $x = (x_n)$, $y_n = \lambda_n x_n$, where $(\lambda_n)$ is a bounded sequence on $\R$, and $a = \inf \lambda_j, b = \sup \lambda_j$.
Show that each $\lambda_j$ is an eigenvalue of $T$.
Under what condition will $[a, b] \subseteq \sigma(T)$.
\end{exercise}
\begin{proof}
For $\lambda_j$, consider $x = (0, \dots, 0, x_j, 0, \dots)$.
We have $Tx = (0, \dots, 0, \lambda_j x_j, 0, \dots) = \lambda_j x$, so that $\lambda_j$ is an eigenvalue.
For $[a, b]$ to be contained in the spectrum of $T$, we need $(\lambda_n)$ to be dense in $[a, b]$.
\end{proof} 

\begin{exercise}{8}
Using Theorem 9.1-2, prove that the spectrum of the operator $T$ in exercise 7 is the closure of the set of eigenvalues.
\end{exercise}
\begin{proof}
We will prove this by double containment.
We have that the eigenvalues are in the spectrum and that the spectrum is closed, so that the closure of the eigenvalues is in the spectrum itself;
that is $\overline{\set{\lambda_n}} \subseteq \sigma(T)$.
We will prove the other direction by contrapositive.
Thus, let $\mu$ be an element outside of the closure of the eigenvalues so that there exists a $\epsilon > 0$, such that $\absoluteValue{\mu - \lambda_i} > \epsilon$.
But this means that $(Sx)_j = x_j/(\mu - \lambda_j)$ (the $j$'th element of the sequence defined by the map) is bounded and well defined for all $j$.
Thus $\mu$ is in the resolvent, giving us the desired result.
\end{proof} 

\begin{exercise}{9}
From 2.2-7 and 3.1-5 we know that the Hilbert space $L^2 [0, 1]$ is the completion of the inner product space $X$ of all continuous functions on $[0, 1]$ with inner product defined by $\brackets{x, y} = \int_0^1 x(t) \overline{y(t)} dt$.
Show that $T: L^2 [0, 1] \to L^2 [0, 1]$ defined by $y(t) = T x(t) = t x(t)$ is a bounded self-adjoint linear operator without eigenvalues.
\end{exercise}
\begin{proof}
$T$ is self-adjoint because 
\begin{align*}
    \brackets{Tx, y} 
    = \int_0^1 t x(t) \overline{y(t)} dt 
    = \int_0^1 x(t) \overline{t y(x)} dt 
    = \brackets{x, Ty}.
\end{align*}

To see $T$ is bounded, note
\begin{align*}
    \norm{T x}^2
    = \brackets{Tx, Tx}
    = \int_0^1 t x(t) \overline{t x(t)} dt
    = \int_0^1 t^2 \absoluteValue{x(t)}^2 dt
    \leq \int_0^1 \absoluteValue{x(t)}^2 dt
    = \norm{x}^2.
\end{align*}

Furthermore, from 7.3-4, we know $\sigma(T) \subset [0, 1]$, since $\norm{T} \leq 1$.
Now consider $Tx - \lambda x = (t - \lambda) x(t) = 0$.
This equality holds only whenever $x(t) = 0$ for all $t$, so that $\lambda$ is not an eigenvalue.
\end{proof} 

\begin{exercise}{10 (Hoellinger-Topelitz)}
It is interesting to note that a linear operator $T$ which is defined on all of a complex Hilbert space $H$ and satisfies $\brackets{Tx, y} = \brackets{x, Ty}$ for all $x, y \in H$ must be bounded (so that our assumption of boundedness at the beginning of the text would not be necessary).
Prove this fact.
\end{exercise}
\begin{proof}
Let $H \times H$ be the direct product of $H$ with itself with the norm being the max between the norm in $H$ for the two arguments.
From the closed graph Theorem (4.13-2), if suffices to show that $G_T = \set{(x, Tx) \in H \times H}$, the graph of $T$, is closed.

Let $((x_n, T x_n))$ be a sequence in $G_T$ converging to $(x, y) \in H \times H$.
We have $x_n \to x$ and $Tx_n \to y$ (sketch: suppose this is not the case and notice we cannot make the max between the norms arbitrarily small).
We prove that $Tx = y$.
Let $z \in H$, we have $\brackets{Tx_n, z} = \brackets{x_n, Tz}$, we have that $\brackets{y, z} = \brackets{x, Tz}$ as $n \to \infty$.
Furthermore, we have $\brackets{x, Tz} = \brackets{Tx, z}$, so that $\brackets{Tx - y, z} = 0$ for all $z$.
Hence $Tx = y$ (3.8-2);
that is, $G_T$ is closed and by the closed graph Theorem, $T$ is bounded.
\end{proof}
