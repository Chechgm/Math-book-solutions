\begin{exercise}{1}
Verify $T = \int_{m-0}^M \lambda d E_\lambda$ for the zero operator $T = 0: H \to H$.
\end{exercise}
\begin{proof}
We have
\begin{align*}
    \int_{0 - 0}^0 \lambda d E_\lambda 
    = 0 E_0 + 0 E_{0 - 0} = 0 I
\end{align*}
\end{proof}

\begin{exercise}{4}
If an operator $T: \R^3 \to \R^3$ is represented, with respect to an orthonormal basis, by a matrix
\begin{align*}
    \begin{bmatrix}
        0 & 1 & 0\\
        1 & 0 & 0\\
        0 & 0 & 1
    \end{bmatrix},
\end{align*}
what is the corresponding spectral family?
Using the result, verify $T = \int_{m-0}^M \lambda dE_\lambda$ for this operator.
\end{exercise}
\begin{proof}
The eigenvalues of $T$ are 1 and $-1$.
We will compute $B_\lambda$, $T_\lambda^+$ and $\cN(T_\lambda^+)$.
First, 
\begin{align*}
    B_1 =
    \begin{bmatrix}
        2 & -2 & 0\\
        -2 & 2 & 0\\
        0 & 0 & 0
    \end{bmatrix}^{1/2}
    =
    \begin{bmatrix}
        1 & -1 & 0\\
        -1 & 1 & 0\\
        0 & 0 & 0
    \end{bmatrix},
\end{align*}
which we can verified with $\brackets{Tx, x}$ with a generic $x \in \R^3$.
Thus, $T_1^+ = \frac{1}{2}[B_1 + T_1] = 0$ and $\cN(T^+_1) = \R^3$.
\end{proof}
