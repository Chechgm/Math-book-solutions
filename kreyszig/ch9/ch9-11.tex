\begin{exercise}{1}
What can we conclude from Theorem 9.11-1 in the case of a Hermitian matrix?
\end{exercise}
\begin{proof}
Since matrices have a finite number of eigenvalues, then $E_\lambda$ for a Hermitian matrix has a finite number of discontinuities.
\end{proof} 

\begin{exercise}{2}
If $T$ in Theorem 9.11-1 is compact and has infinitely many eigenvalues, what can we conclude about $(E_k)$ from Theorems 9.11-1 and 9.11-2?
\end{exercise}
\begin{proof}
From 9.11-1 and 9.11-2, we can conclude that $E_\lambda$ is a step function with countably many jumps.
In particular, for $\lambda < \norm{T}$, we have $E_\lambda = 0$ and for $\lambda \geq \norm{T}$, we have $E_\lambda = I$.
\end{proof}

\begin{exercise}{4}
We know that if $m$ in Theorem 9.2-1 is positive, then $T$ is positive.
How does this follow from the spectral representation $T = \int_{m-0}^M \lambda dE_\lambda$ in section 9.9?
\end{exercise}
\begin{proof}
We write $\brackets{Tx, x} = \int_{m-0}^M \lambda d w(\lambda)$, where $w(\lambda) = \brackets{E_\lambda x, x}$.
Since $\sigma(T) \subseteq [m, M]$, then $E_\lambda = 0$ for $\lambda \leq 0$ so that $\int_{m-0}^M \lambda d w(\lambda) \geq \int_{m-0}^M d w(\lambda) = m \brackets{x, x} \geq 0$, so that $T$ is positive.
\end{proof} 

\begin{exercise}{5}
We know that the spectrum of a bounded self-adjoint linear operator is closed.
How does this follow from theorems in this section?
\end{exercise}
\begin{proof}
Note that 9.11-2 tells us that $\lambda_0 \in \rho(T)$, if and only if there exists $\gamma > 0$ such that $E_\lambda$ is constant in $[\lambda_0 - \gamma, \lambda_0 + \gamma]$ but note this defines an open set, since we can take any other point in the interval, and it will still satisfy the same property.
Thus $\rho(T)$ is an open set so that $\sigma(T)$ is closed.
\end{proof}

\begin{exercise}{7 (Pure point spectrum)}
A bounded self-adjoint linear operator $T: H \to H$ on a Hilbert space $H \neq \set{0}$ is said to have a pure point spectrum or purely discrete spectrum if $T$ has an orthonormal set of eigenvectors which is total in $H$.
Illustrate with an example that this does not imply $\sigma_c(T)$ (so that this terminology, which is generally used, may confuse the beginner for a moment).
\end{exercise}
\begin{proof}
An example would be $T: \ell^2 \to \ell^2$ given by $(x_1, x_2, \dots) \mapsto (x_1, x_2/2, x_3/3, \dots)$.
We have proved that the eigenvalues of $T$ are $(e_n)$ which is a total orthonormal set in $\ell^2$, but also $0 \in \sigma_p(T)$, as we proved in exercise 9.7.2.
\end{proof} 

\begin{exercise}{8}
Give examples of compact self-adjoint linear operators $T: \ell^2 \to \ell^2$ having a pure point spectrum such that the set of the nonzero eigenvalues 
\begin{enumerate}
    \item is a finite point set,
    \item is an infinite point set and the corresponding eigenvectors form a dense set in $\ell^2$,
    \item is an infinite point set and the corresponding eigenvectors span a subspace of $\ell^2$ such that the orthogonal complement of the closure of that subspace is finite dimensional,
    \item as in the previous numeral but that complement is infinite dimensional.
In each case find a total orthonormal set of eigenvectors.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item $I: \ell^2 \to \ell^2$ has $\set{1} = \sigma_p(I)$.
    \item fill
    \item Consider $T: \ell^2 \to \ell^2$ given by $(x_1, x_2, x_3, \dots) \mapsto (x_1, x_1, x_3/3, \dots )$.
    We have as eigenvectors $(e_n)_{n=3}^\infty$, and $(1, 1, 0, \dots)$.
    The orthogonal complement of the subspace spanned by these eigenvectors is $\vecspan\set{(1, -1, 0, \dots)}$, which is finite dimensional.
    \item We can use a similar technique as in the previous numeral, by defining the operator $(x_1, x_2, \dots) \mapsto (x_1, x_1, x_3/3, x_4/4, x_4/4, x_6/6, x_7/7, x_7/7, \dots)$.
    We will obtain a set of eigenvectors given by $(e_{3n})_{n=1}^\infty$ and $\set{(1, 1, 0, \dots), (0, 0, 0, 1, 1, 0, \dots), \dots}$ whose span closure has as orthogonal complement the subspace spanned by $\set{(1, -1, 0, \dots), (0, 0, 0, 1, -1, 0, \dots), \dots}$, which is infinite dimensional.
\end{enumerate}
\end{proof} 

\begin{exercise}{9 (Purely continuous spectrum)}
A bounded self-adjoint linear operator $T: H \to H$ on a Hilbert space $H \neq \set{0}$ is said to have a purely continuous spectrum if $T$ has no eigenvalues.
If $T$ is any bounded self-adjoint operator on $H$, show that there is a closed subspace $Y \subseteq H$ which reduces $T$ (see exercise 9.6.10) and is such that $T_1 = T|_Y$ has a pure point spectrum, whereas $T_2 = T|_Z, Z = ^\perp$ has a purely continuous spectrum.
(This reduction facilitated the investigation of $T$).
\end{exercise}
\begin{proof}
Take the closure of the span of the eigenvectors of $T$ in $H$.
This space is closed by definition and the set of eigenvectors is total in it.
If the eigenvectors are not orthonormal, we can make them orthonormal through the Gram-Schmidt procedure.
The complement of this space is orthogonal to it, and by construction it has no eigenvalues, as required.
\end{proof} 

\begin{exercise}{10}
What can we say about the spectral families $(E_{\lambda 1})$ and $(E_{\lambda 2})$ of $T_1$ and $T_2$ in exercise 9 in terms of the spectral family $(E_\lambda)$ of $T$?
\end{exercise}
\begin{proof}
The spectral family of the subspace with the pure point spectrum would be a step function between 0 and $I$, where the jump discontinuities happen at the eigenvalues.
On the other hand, the spectrum of the purely continuous subspace is a continuous monotonic function between 0 and $I$.
Of course these are derived from the original spectral family.
\end{proof}
