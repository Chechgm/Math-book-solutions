\begin{exercise}{3}
Find $m = \inf_{\norm{x}=1} \brackets{Tx, x}$ and $M = \sup_{\norm{x}=1} \brackets{Tx, x}$ (Theorem 9.2-1) if $T$ is the projection operator of a Hilbert space $H$ onto a proper subspace $Y \neq \set{0}$ of $H$.
\end{exercise}
\begin{proof}
Since $T$ is a projection, then for any $x \in H$, we can write $x = y + y'$, where $y \in Y$ and $y' \in Y^\perp$.
Thus, we have 
\begin{align*}
    \brackets{Tx, x}
    = \brackets{T(y + y'), y + y'}
    = \brackets{y, y + y'}
    = \norm{y}^2,
\end{align*}
since $y$ and $y'$ are orthogonal to each other.
If $\norm{x} = 1$, then $\norm{x}^2 = \norm{y}^2 + \norm{y'}^2$ (by the Pythagorean Theorem), so that $\norm{y}^2 \leq 1$.
Thus, $M = 1$ since the supremum is attained whenever $x \in Y$ and $m = 0$ since the infimum is attained whenever $x \in Y^\perp$.
\end{proof} 

\begin{exercise}{4}
Prove that $m \in \sigma(T)$ in Theorem 9.2-3.
\end{exercise}
\begin{proof}
This is a Corollary to Theorem 9.2-3 by noting that $m \in \sigma(T)$ if and only if $-m \in \sigma(-T)$ by the spectral mapping Theorem for polynomials (7.4-2).
We then know that $-m = \sup_{\norm{x} = 1} \brackets{-Tx, x}$ by the usual properties of the supremum and infimum.
Then Theorem 9.2-3 tells us that $-m$ is in $\sigma(T)$ so that $m \in \sigma(T)$, completing the proof.
\end{proof}

\end{exercise}
\begin{proof}
This is a consequence of 9.2.3, since $m$ and $M$ belong to the spectrum of the operator.
\end{proof} 

\begin{exercise}{6}
Show that a compact self-adjoint linear operator $T: H \to H$ on a complex Hilbert space $H \neq \set{0}$ has at least one eigenvalue.
\end{exercise}
\begin{proof}
We know that for all $\lambda \in \sigma(T)$ (besides 0) are eigenvalues in a compact operator (8.4-4).
Next, we know from 9.2-3, that $m = \inf_{\norm{x} = 1} \brackets{Tx, x}$ and $M = \sup_{\norm{x} = 1} \brackets{Tx, x}$ are both part of the spectrum of $T$.
Thus, as long as $m, M \neq 0$ then $\sigma_p(T)$ is not empty.
The only case in which both $M$ and $m$ would be zero is if $T = 0$, giving us the desired result.
\end{proof} 

\begin{exercise}{7}
Consider the operator $T: \ell^2 \to \ell^2$ defined by $y = T x$, where $x = (x_n)$, $y = (y_n)$ and $y_n = x_n/n$, for $n = 1, 2, \dots$.
It was shown in 8.1-6 that $T$ is compact.
Find the spectrum of $T$.
Show that $0 \in \sigma_c(T)$ and, actually, $\sigma_c(T) = \set{0}$.
(For compact operators with $0 \in \sigma_p(T)$ or $0 \in \sigma_r(T)$.
See exercises 4 and 5 in section 8.4).
\end{exercise}
\begin{proof}
Note that $T$ is self- adjoint, since for any $x, y \in \ell^2$, we have $\brackets{Tx, y} = \sum_i (x_i / i) \bar{y_i} = \brackets{x, Ty}$.
Furthermore, we have that $\sup_{\norm{x} = 1} \brackets{Tx, x} = 1$ (choose $x = (1, 0, \dots)$ and $\inf_{\norm{x} = 1} \brackets{Tx, x} = 0$, (consider the sequence of sequences, where $x_n$ is given by $x_n = (0, \dots, 0, 1, 0, \dots)$, the 1 appearing in the $n$'th position).
Thus, the spectrum of $T$ lies in $[0, 1]$, and furthermore, $0, 1 \in \sigma(T)$.
Moreover, $0 \notin \sigma_p(T)$, if this were the case, $Tx = 0$ would have a nontrivial solution, which is not the case.
Finally, 9.2-4 tells us that $\sigma_r(T)$ is empty.
Putting everything together, we get that $0 \in \sigma_c(T)$.
\end{proof} 

\begin{exercise}{9}
If $\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n$ are the eigenvalues of a Hermitian matrix $A$, show that $\lambda_1 \max_{x \neq 0} q(x)$, $\lambda_n = \min_{x \neq 0} q(x)$, where $q(x) = (\bar{x}^T A x) / (\bar{x}^T x)$.
Show that, furthermore, $\lambda_j = \max_{x \in Y_j, x \neq 0} q(x)$, for $j = 2, 3, \dots$, where $Y_j$ is the subspace of $\C^n$ consisting of all vectors that are orthogonal to the eigenvectors corresponding to $\lambda_1, \dots, \lambda_{j-1}$.
\end{exercise}
\begin{proof}
We have 
\begin{align*}
    \brackets{Ax, x} 
    =& \brackets{x, Ax}\\
    =& x^T (\overline{Ax})\\ 
    =& [x^T (\overline{A x})]^T \\
    =& \bar{x}^T \bar{A}^T
    = x^T A x,
\end{align*}
and
\begin{align*}
    \frac{\brackets{Ax, x}}{\brackets{x, x}}
    = \frac{\brackets{Ax, x}}{\norm{x}^2}
    = \brackets{Ax / \norm{x}^2, x / \norm{x}^2},
\end{align*}
so that taking the supremum (and infimum) over all $x \neq 0$ is equivalent to taking the supremum (and infimum) of $\brackets{Ax, x}$ over all $x$ with $\norm{x} = 1$.
In other words, $M = \sup_{x \neq 0} q(x)$ and $m = \inf_{x \neq 0} q(x)$.
Theorem 9.2-3 tells us that $m = \lambda_n$ and $M = \lambda_1$.

Now let $x \in Y_j$.
Then $\brackets{x, e_i} = 0$ for an eigenvector corresponding to $\lambda_i$.
Since $Y_j$ is invariant under $A$, then the previous result implies that $\lambda_j = \sup_{x \in Y_j, x \neq 0} q(x)$.
\end{proof} 

\begin{exercise}{10}
Show that a real symmetric square matrix $A$ with positive elements has a positive eigenvalue.
(It can be shown that the statement holds even without the assumption of symmetry;
this is part of a famous Theorem by Perron and Frobenius).
\end{exercise}
\begin{proof}
Notice that a real symmetric square matrix is hermitian because it equals its conjugate transpose.
Moreover, consider $x^T = (1, 0, \dots, 0)$.
Then $\bar{x}^T A x =  a_{1, 1} \geq 0$, so that $\lambda_j = \sup_{x \neq 0} \brackets{Ax, x} / \brackets{x, x} \geq a_i$, and thus $A$ has a positive real eigenvalue.
\end{proof} 

\begin{exercise}{Extra (Spectral Theorem for compact self-adjoint operators)}
Let $T$ be a self-adjoint compact operator on a separable Hilbert space, let $\absoluteValue{\lambda_1} \geq \absoluteValue{\lambda_2} \geq \dots$ be the nonzero eigenvalues of $T$ with corresponding orthonormal eigenvectors $(u_k)$, then $(u_k)$ is an orthonormal basis of $\overline{\range(T)}$. 
If we let $(v_k)$ be a basis for the kernel of $T$, then $(u_k) \cup (v_k)$ is an orthonormal basis of $H$. 
Deduce then that we can find a basis of $H$ consisting of eigenvectors, which is essentially a perfect reproduction of the finite dimensional spectral theorem.
\end{exercise}
\begin{proof}
Suppose that a list of orthonormal eigenvectors $(u_1, \dots u_k)$ has already been found.
We can find the next orthonormal vector using the eigenvalue given by $\sup \absoluteValue{\brackets{Tu, u}}$, where the supremum is taken over the orthogonal complement of $U_k = \vecspan (u_1, \dots, u_k)$.
This is true, given that $U_k^\perp$ is itself a Hilbert space in which $T|_{U_k}$ is defined so that its spectrum lies in the interval $[\inf_{\norm{x} = 1} \brackets{Tx, x}, \sup_{\norm{x} = 1} \brackets{Tx, x}]$ (Theorem 9.2-1), both ends of the interval are spectral values of $T$ (Theorem 9.2-3), and eigenvectors corresponding to different eigenvalues of $T$ are orthogonal (Theorem 9.1-1).
Of course eigenvalues and eigenvectors of $T|_{U_k}$ are also eigenvectors and eigenvalues of $T$, so that the found eigenvector is orthogonal to $\vecspan (u_1, \dots, u_k)$.

If $T$ has finite rank, the process finishes at some point

\end{proof}
