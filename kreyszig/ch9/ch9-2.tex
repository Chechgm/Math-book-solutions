\section{Further spectral properties of bounded self-adjoint linear operators}


\begin{exercise}{2}
What Theorem about the eigenvalues of a Hermitian matrix $A$ do we obtain from Theorem 9.2-1?
\end{exercise}
\begin{proof}
We get that the eigenvalues of a Hermitian matrix are real.
\end{proof} 

\begin{exercise}{3}
Find $m = \inf_{\norm{x}=1} \brackets{Tx, x}$ and $M = \sup_{\norm{x}=1} \brackets{Tx, x}$ (Theorem 9.2-1) if $T$ is the projection operator of a Hilbert space $H$ onto a proper subspace $Y \neq \set{0}$ of $H$.
\end{exercise}
\begin{proof}
Since $T$ is a projection, then for any $x \in H$, we can write $x = y + y'$, where $y \in Y$ and $y' \in Y^\perp$.
Thus, we have 
\begin{align*}
    \brackets{Tx, x}
    = \brackets{T(y + y'), y + y'}
    = \brackets{y, y + y'}
    = \norm{y}^2,
\end{align*}
since $y$ and $y'$ are orthogonal to each other.
If $\norm{x} = 1$, then $\norm{x}^2 = \norm{y}^2 + \norm{y'}^2$ (by the Pythagorean Theorem), so that $\norm{y}^2 \leq 1$.
Thus, $M = 1$ since the supremum is attained whenever $x \in Y$ and $m = 0$ since the infimum is attained whenever $x \in Y^\perp$.
\end{proof} 

\begin{exercise}{4}
Prove that $m \in \sigma(T)$ in Theorem 9.2-3.
\end{exercise}
\begin{proof}
This is a Corollary to Theorem 9.2-3 by noting that $m \in \sigma(T)$ if and only if $-m \in \sigma(-T)$ by the spectral mapping Theorem for polynomials (7.4-2).
We then know that $-m = \sup_{\norm{x} = 1} \brackets{-Tx, x}$ by the usual properties of the supremum and infimum.
Then Theorem 9.2-3 tells us that $-m$ is in $\sigma(T)$ so that $m \in \sigma(T)$, completing the proof.
\end{proof} 

\begin{exercise}{5}
Show that the spectrum of a bounded self-adjoint linear operator on a complex Hilbert space $H \neq \set{0}$ is not empty, using one of the theorems of the present section.
\end{exercise}
\begin{proof}
This is a consequence of 9.2.3, since $m$ and $M$ belong to the spectrum of the operator.
\end{proof} 

\begin{exercise}{6}
Show that a compact self-adjoint linear operator $T: H \to H$ on a complex Hilbert space $H \neq \set{0}$ has at least one eigenvalue.
\end{exercise}
\begin{proof}
We know that for all $\lambda \in \sigma(T)$ (besides 0) are eigenvalues in a compact operator (8.4-4).
Next, we know from 9.2-3, that $m = \inf_{\norm{x} = 1} \brackets{Tx, x}$ and $M = \sup_{\norm{x} = 1} \brackets{Tx, x}$ are both part of the spectrum of $T$.
Thus, as long as $m, M \neq 0$ then $\sigma_p(T)$ is not empty.
The only case in which both $M$ and $m$ would be zero is if $T = 0$, giving us the desired result.
\end{proof} 

\begin{exercise}{7}
Consider the operator $T: \ell^2 \to \ell^2$ defined by $y = T x$, where $x = (x_n)$, $y = (y_n)$ and $y_n = x_n/n$, for $n = 1, 2, \dots$.
It was shown in 8.1-6 that $T$ is compact.
Find the spectrum of $T$.
Show that $0 \in \sigma_c(T)$ and, actually, $\sigma_c(T) = \set{0}$.
(For compact operators with $0 \in \sigma_p(T)$ or $0 \in \sigma_r(T)$.
See exercises 4 and 5 in section 8.4).
\end{exercise}
\begin{proof}
Note that $T$ is self- adjoint, since for any $x, y \in \ell^2$, we have $\brackets{Tx, y} = \sum_i (x_i / i) \bar{y_i} = \brackets{x, Ty}$.
Furthermore, we have that $\sup_{\norm{x} = 1} \brackets{Tx, x} = 1$ (choose $x = (1, 0, \dots)$ and $\inf_{\norm{x} = 1} \brackets{Tx, x} = 0$, (consider the sequence of sequences, where $x_n$ is given by $x_n = (0, \dots, 0, 1, 0, \dots)$, the 1 appearing in the $n$'th position).
Thus, the spectrum of $T$ lies in $[0, 1]$, and furthermore, $0, 1 \in \sigma(T)$.
Moreover, $0 \notin \sigma_p(T)$, if this were the case, $Tx = 0$ would have a nontrivial solution, which is not the case.
Finally, 9.2-4 tells us that $\sigma_r(T)$ is empty.
Putting everything together, we get that $0 \in \sigma_c(T)$.
\end{proof} 

\begin{exercise}{9}
If $\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n$ are the eigenvalues of a Hermitian matrix $A$, show that $\lambda_1 \max_{x \neq 0} q(x)$, $\lambda_n = \min_{x \neq 0} q(x)$, where $q(x) = (\bar{x}^T A x) / (\bar{x}^T x)$.
Show that, furthermore, $\lambda_j = \max_{x \in Y_j, x \neq 0} q(x)$, for $j = 2, 3, \dots$, where $Y_j$ is the subspace of $\C^n$ consisting of all vectors that are orthogonal to the eigenvectors corresponding to $\lambda_1, \dots, \lambda_{j-1}$.
\end{exercise}
\begin{proof}
We have 
\begin{align*}
    \brackets{Ax, x} 
    =& \brackets{x, Ax}\\
    =& x^T (\overline{Ax})\\ 
    =& [x^T (\overline{A x})]^T \\
    =& \bar{x}^T \bar{A}^T
    = x^T A x,
\end{align*}
and
\begin{align*}
    \frac{\brackets{Ax, x}}{\brackets{x, x}}
    = \frac{\brackets{Ax, x}}{\norm{x}^2}
    = \brackets{Ax / \norm{x}^2, x / \norm{x}^2},
\end{align*}
so that taking the supremum (and infimum) over all $x \neq 0$ is equivalent to taking the supremum (and infimum) of $\brackets{Ax, x}$ over all $x$ with $\norm{x} = 1$.
In other words, $M = \sup_{x \neq 0} q(x)$ and $m = \inf_{x \neq 0} q(x)$.
Theorem 9.2-3 tells us that $m = \lambda_n$ and $M = \lambda_1$.

Now let $x \in Y_j$.
Then $\brackets{x, e_i} = 0$ for an eigenvector corresponding to $\lambda_i$.
Since $Y_j$ is invariant under $A$, then the previous result implies that $\lambda_j = \sup_{x \in Y_j, x \neq 0} q(x)$.
\end{proof} 

\begin{exercise}{10}
Show that a real symmetric square matrix $A$ with positive elements has a positive eigenvalue.
(It can be shown that the statement holds even without the assumption of symmetry;
this is part of a famous Theorem by Perron and Frobenius).
\end{exercise}
\begin{proof}
Notice that a real symmetric square matrix is hermitian because it equals its conjugate transpose.
Moreover, consider $x^T = (1, 0, \dots, 0)$.
Then $\bar{x}^T A x =  a_{1, 1} \geq 0$, so that $\lambda_j = \sup_{x \neq 0} \brackets{Ax, x} / \brackets{x, x} \geq a_i$, and thus $A$ has a positive real eigenvalue.
\end{proof} 

\begin{exercise}{Extra (Spectral Theorem for compact self-adjoint operators)}
Let $T$ be a self-adjoint compact operator on a separable Hilbert space, let $\absoluteValue{\lambda_1} \geq \absoluteValue{\lambda_2} \geq \dots$ be the nonzero eigenvalues of $T$ with corresponding orthonormal eigenvectors $(x_k)$, then $(x_k)$ is an orthonormal basis of $\overline{\range(T)}$. 
If we let $(v_k)$ be a basis for the kernel of $T$, then $(x_k) \cup (v_k)$ is an orthonormal basis of $H$. 
Deduce then that we can find a basis of $H$ consisting of eigenvectors, which is essentially a perfect reproduction of the finite dimensional spectral theorem.
\end{exercise}
\begin{proof}
We will adapt a proof due to Casey Rodriguez on MIT's introduction to functional analysis course notes, including the results we have derived in Kreyszig and details whenever we feel they are needed.

We start by proving the \textbf{maximum principle}:
Let $T$ be a self-adjoint compact operator.
We can order the nonzero eigenvalues of $T$ as $\absoluteValue{\lambda_1} \geq \absoluteValue{\lambda_2} \geq \dots$ (including multiplicity), such that we have pairwise orthonormal eigenvectors $\set{u_k}$ for $\lambda_k$ satisfying 
\begin{align*}
    \absoluteValue{\lambda_j} 
    = \sup_{\norm{x} =1, x \in \vecspan(x_1, \dots, x_{j-1})^\perp} 
    = \absoluteValue{\brackets{Tx, x}} = \absoluteValue{\brackets{T x_j, x_j}}.
\end{align*}
Furthermore, we have $\absoluteValue{\lambda_j} \to 0$ as $j \to \infty$ if the sequence of nonzero eigenvalues does not terminate.

\textit{Proof}.
Since $T$ is compact, it has countably many eigenvalues (8.3-1), and by the same Theorem, the only potential point of accumulation for the eigenvalues is 0, so that the second part of the maximum principle has been already been proven. 
Furthermore, 8.3-3 tells us that the eigenspace of each eigenvalue is finite.

We will show that the equation holds by constructing our eigenvalues inductively.
The base case is the content of Theorem 9.2-3, were we proved that $m = \inf_{\norm{x}=1} \brackets{Tx, x}$ and $M = \sup_{\norm{x}=1} \brackets{Tx, x}$ are spectral values (in the case of compact operators, eigenvalues) of $T$.
Furthermore, we know that eigenvectors of numerically distinct eigenvalues of self-adjoint operators are orthogonal (9.1-1.b), we can normalise the eigenvectors to form orthonormal eigenvectors.

For the inductive step, suppose we have found $\lambda_1, \dots, \lambda_n$ and their corresponding orthonormal eigenvectors $x_1, \dots, x_n$ satisfying the equation of the maximum principle. 
We explore two cases, in the first case we have $Tx = \sum_{k=1}^n \lambda_k \brackets{x, x_k} u_k$, meaning we found all of the eigenvalues and process terminates (because $T$ is a finite-rank operator).
In the case where $T$ is not finite-rank, the equation does not hold so in order to find $\lambda_{n+1}$, we can define a linear operator $T_n$ via $T_n x = Tx - \sum_{k=1}^n \lambda_k \brackets{x, x_k} x_k$.
$T_n$ is a self-adjoint compact operator (since $T$ is self-adjoint, the $\lambda_k$ are real values by 9.1-1.a, and $T_n$ is a sum of a compact operator $T$ and a finite-rank operator so that it is itself compact by exercise 8.1.2).
Thus, if $x \in \vecspan(x_1, \dots x_n)$, then 
\begin{align*}
    T_n x 
    = T(\sum_{k=1}^n a_k x_k) - \sum_{k=1}^n \lambda_k \brackets{x, x_k} x_k
    = \sum_{k=1}^n a_k T(x_k) - \sum_{k=1}^n \lambda_k a_k x_k
    = 0,
\end{align*}
since $Tx_k = \lambda_k x_k$.
Furthermore, for any $x \in \vecspan(x_1, \dots, x_n)^\perp$, we have $T_n x = Tx$ because the sum term is 0.

Therefore, for any $x \in H$ and any $v \in \vecspan(x_1, \dots x_n)$, we have $\brackets{T_n x, v} = \brackets{x, T_n v} = 0$ (the first step due to the self-adjointness of $T_n$ and the second step because our reasoning above).
This is the same as saying that $T_n x$ is always in the orthogonal complement of $\vecspan(x_1, \dots x_n)$, so $\range(T_n) \subseteq \vecspan(x_1, \dots x_n)^\perp$.
From this fact, we learn that if $T_n x = \lambda x$ for some nonzero $\lambda x$, then $x = T_n(x / \lambda)$ is in the range of $T_n$, so it is in $\vecspan(x_1, \dots x_n)^\perp$.
From our previous conclusions, this means that $T_n x = Tx = \lambda x$, so any nonzero eigenvalue of $T_n$ is also a nonzero eigenvalue of $T$.

Thus, we can apply 9.2-3 again to conclude that $T_n$ has a nonzero eigenvalue $\lambda_{n+1}$ with unit eigenvector $x_{n+1}$ orthogonal to the span of $(x_1, \dots x_n$) (since $T_n$ is zero on that span), with $\absoluteValue{\lambda_{n+1}} = \sup_{\norm{x}=1} \absoluteValue{\brackets{T_n x, x}}$.

Since we are still working in the same Hilbert space, this expression can be written in terms of $T$ as well.
To see this is the case, note that 
\begin{align*}
    \absoluteValue{\lambda_{n+1}}
    = \sup_{\norm{x}=1, x \in \vecspan(x_1, \dots, x_n)^\perp} \absoluteValue{\brackets{T_n x, x}},
\end{align*}
since $T_n x$ is zero on $\vecspan(x_1, \dots, x_n)$, and then when we restrict to those $x$, we have $T_n x = Tx$ so the expression above equals $\sup_{\norm{x}=1, x \in \vecspan(x_1, \dots, x_n)^\perp} \absoluteValue{\brackets{T x, x}}$, which gives us the desired equation.
We also preserve the ordering of the eigenvalues because 
\begin{align*}
    \absoluteValue{\lambda_{n+1}}
    =& \sup_{\norm{x}=1, x \in \vecspan(x_1, \dots, x_n)^\perp} \absoluteValue{\brackets{T x, x}}\\
    \leq& \sup_{\norm{x}=1, x \in \vecspan(x_1, \dots, x_{n-1})^\perp} \absoluteValue{\brackets{T x, x}}
    =\absoluteValue{\lambda_n}.
\end{align*}
Finally because $\absoluteValue{\lambda_{n+1}} = \absoluteValue{\brackets{Tx_{n+1}, x_{n+1}}}$, we have shown all the desired results, and finished the proof of the maximum principle $\square$.

We will now prove the spectral theorem.
First, note that the process described in the proof of the maximum principle terminates if and only if $T$ has finite rank, meaning that there is some $n$ with $Tx = \sum_{k=1}^n \lambda_k \brackets{x, x_k} x_k$.
In such case, $\range(T) \subseteq \vecspan(x_1, \dots, x_k)$, and thus $\set{x_k}$ do indeed form an orthonormal basis for $\range(T)$ and also of $\overline{\range(T))}$.

Otherwise, the process does not terminate, and thus we have countably infinitely many nonzero eigenvalues $(\lambda_n)$ counter with multiplicity.
We know that $\absoluteValue{\lambda_k} \to 0$, and that the $x_k$ form an orthonormal subset of $\range(T)$.
To show this is a basis, we must show that if $u \in \range(T)$ and $\brackets{u, x_k} = 0$ for all $k$, then $u = 0$.
To see this is the case, we first write $u = Tx$ for some $x \in H$, meaning that $\brackets{Tx, x_k} = 0$ for all $k$.
Since $T$ is self-adjoint, this means (because $\lambda_k$ is real) that $\lambda_k \brackets{x, x_k} = \brackets{x, \lambda_k x_k} = \brackets{Tx, x_k} = 0$.
Therefore $x$ is orthogonal to all $x_k$, so by the maximum principle, 
\begin{align*}
    \norm{u} 
    = \norm{Tx} 
    = \norm{(A - \sum_{k=1}^n \lambda_k \brackets{x, x_k} x_k) x}
\end{align*}
(since each term in the sum is zero), meaning that we can write this as $\norm{T_n x} \leq \absoluteValue{\lambda_{n+1}} \norm{x}$.
Taking $n \to \infty$ and noting that the $\lambda$ converge to 0, we must have $\norm{u} = 0$.
This proves that the eigenvectors indeed form an orthonormal basis for the range of $T$.

To show that they also form an orthonormal basis for the closure of the range, notice that $\overline{\range(T)} \subseteq \overline{\vecspan(\set{x_k}})$ and that the closure of the span can be written as $\set{\sum_k a_k x_k: \sum_k \absoluteValue{a_k}^2 < \infty}$.
Therefore $\set{x_k}$ must indeed be an orthonormal basis for $\overline{\range(T)}$.
This means we have an orthonormal basis of $\overline{\range(T)} = (\range(T)^\perp)^\perp = (\nullspace(T))^\perp$, so to complete the orthonormal basis of $H$, we need an orthonormal basis of $\nullspace(T)$, which exists because $H$ is separable and thus $\nullspace(T)$ is also separable (3.2-4.c).
\end{proof}
