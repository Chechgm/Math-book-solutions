\begin{exercise}{1}
Let $T: \cD(T) \to \ell^2$, where $\cD(T) \subseteq \ell^2$ consists of all $x = (x_n)$ with only finitely many nonzero terms $x_n$ and $y = (y_n) = Tx = (n x_n)$.
This operator is unbounded (exercise 10.1.8).
Show that $T$ is not closed.
\end{exercise}
\begin{proof}
Consider the sequence $(x_n)$ where each $x_n \in \cD(T)$ is defined as $x_n = (1, 1/2^2, 1/3^2, \dots, 1/n^2, 0, \dots)$.
We have $x_n \to x = (1, 1/2^2, 1/3^, \dots)$, which is not in $\cD(T)$, since it has infinitely many nonzero entries.
Furthermore, $Tx = (1, 1/2, 1/3, \dots)$, which does not even belong to $\ell^2$, since the series associated to the sequence does not converge.
Thus, $T$ is not closed.
\end{proof} 

\begin{exercise}{4}
Let $T: \cD(T) \to H$ be a closed linear operator.
If $T$ is injective, show that $T^{-1}$ is closed.
\end{exercise}
\begin{proof}
Since $T$ is injective, then we can identify any pair $(x, Tx) = (x, y) \in \cG(T)$ with a pair $(y, x) = (Tx, T^{-1}(Tx)) \in \cG(T^{-1})$ (there are no two $x \in \cD(T)$ that map into $y$).
Since $\cG(T)$ is closed in $H \times H$, then so is $\cG(T^{-1})$.
\end{proof}

\begin{exercise}{6}
If $T$ is a symmetric linear operator, show that $T^{\ast \ast}$ is a closed symmetric linear extension of $T$.
\end{exercise}
\begin{proof}
This is a combination of some previous results.
We know that $T^\ast$ is closed (10.3-3) and applying the same result again, we can conclude $T^{\ast \ast}$ is closed too (this is even independent of whether $T$ is closed or not).
We also proved that if $T$ is symmetric, then $T^{\ast \ast}$ is symmetric (exercise 10.2.4).
To see $T^{\ast \ast}$ is a linear extension of $T$, note that since $T$ is symmetric $T \subseteq T^\ast$ (10.2-4), and since $T$ is densely defined, in $H$, so is $T^\ast$, so that we can apply 10.2-1.b, which tells us that $T \subseteq T^{\ast \ast}$, completing the proof.
\end{proof} 

\begin{exercise}{7}
Show that the graph $\cG(T^\ast)$ of the Hilbert-adjoint operator of a linear operator $T$ is related to $\cG(T)$ by $\cG(T^\ast) = [U(\cG(T))]^\perp$, where $U: H \times H \to H \times H$ is defined by $(x, y) \mapsto (y, -x)$.
\end{exercise}
\begin{proof}
First, we will define the inner product on $H \times H$ as $\brackets{(x, y), (u, v)} = \brackets{x, u} + \brackets{y, v}$.
With this, we will prove the statement by double containment:

($\subseteq$)
Let $(y, T^\ast y) \in \cG(T^\ast)$.
We want to prove $0 = \brackets{(Tx, -x), (y, T^\ast y)}$ for all $(x, Tx) \in \cG(T)$.
We have $\brackets{(Tx, -x), (y, T^\ast y)} = \brackets{Tx, y} - \brackets{x, T^\ast y}$, which is precisely 0 by the definition of $T^\ast$.

($\supseteq$)
Let $(y, y^\ast) \in [U(\cG(T))]^\perp$.
We want to show that for all $x \in \cD(T)$, we have $\brackets{Tx, y} = \brackets{x, y^\ast}$, where $y^\ast = T^\ast y$.
We have that for all $(x, Tx)$, it holds that 
\begin{align*}
    0 
    = \brackets{U(x, Tx), (y, y^\ast)} 
    = \brackets{(Tx, -x), (y, y^\ast)}
    = \brackets{Tx, y} - \brackets{x, y^\ast},
\end{align*}
so that $\brackets{Tx, y} = \brackets{x, y^\ast}$, which is precisely what we wanted to prove.
\end{proof}

\begin{exercise}{9 (Closed graph Theorem)}
Show that a closed linear operator $T: H \to H$ on a complex Hilbert space $H$ is bounded.
[Hint: Use exercise 8.
Of course, give an independent proof, without using 4.13-2].
\end{exercise}
\begin{proof}
From the statement, I assume $\cD(T) = H$.
Thus, $T^\ast$ is bounded (exercise 10.1.9) and closed (10.3-3).
This means that $\cD(T^\ast) = H$, and applying exercise 10.1-9 again we get that $T^{\ast \ast}$ is bounded.
We have all the conditions for exercise 8 to hold, so that $T = T^{\ast \ast}$, meaning that $T$ is bounded too.
\end{proof} 

\begin{exercise}{10}
If $T$ is closed, show that $T_\lambda = T - \lambda I$ is closed, and if $T_\lambda^{-1}$ exists, then $T_\lambda^{-1}$ is closed.
\end{exercise}
\begin{proof}
This follows by the continuity of transforming $T$ into $T - \lambda I$:
we have that $x_n \to x$ and $Tx_n \to y = Tx$, with $x \in \cD(T)$ and $Tx = y$ (10.3-2).
Suppose $T_\lambda x_n \to y$, then $T x_n - \lambda x_n \to Tx - \lambda x = T_\lambda x$, so that $T_\lambda$ is closed.

\end{proof}
