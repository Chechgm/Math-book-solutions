\section{Approximation in Hilbert space}


\begin{exercise}{4}
Express the Schwarz inequality in terms of a Gram determinant.
Use Theorem 6.5-1 to obtain the condition under which the equality sign holds.
\end{exercise}
\begin{proof}
Consider the Gram determinant
\begin{align*}
    G(x, y)
    = \begin{vmatrix}
        \brackets{x,x} & \brackets{x,y}\\
        \brackets{y,x} & \brackets{y,y}
    \end{vmatrix}
    = \brackets{x, x}\brackets{y,y} - \brackets{x, y}\brackets{y, x}
    = \norm{x}^2\norm{y}^2 - \absoluteValue{\brackets{x, y}},
\end{align*}
so that the Schwarz inequality can be written as $G(x,y) \geq 0$.
\end{proof}
