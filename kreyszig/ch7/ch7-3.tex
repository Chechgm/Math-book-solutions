\begin{exercise}{1}
Let $X = C[0, 1]$ and define $T: X \to X$ by $Tf = gf$, where $g \in X$ is fixed.
Find $\sigma(T)$.
Note that $\sigma(T)$ is closed.
\end{exercise}
\begin{proof}
We want to find $\lambda$ such that $R_\lambda(T) = (T - \lambda I)^{-1}$ either doesn't exists, is not bounded or is not defined on a dense subset of $X$.
For a fixed $\lambda$, $R_\lambda(T) = f(t) / (g(t) - \lambda)$, so that $R_\lambda(T)$ does not exist if $\lambda \in \cR(g)$.
Furthermore, since $g \in C[0, 1]$, then $g([0,1])$ is compact, so that $\sigma(T)$ is closed.
\end{proof} 

\begin{exercise}{2}
Find a linear operator $T: C[0, 1] \to C[0, 1]$ whose spectrum is a given interval $[a, b]$.
\end{exercise}
\begin{proof}
Based on our previous solution, choose $g \in C([0,1])$ with $\cR(g) = [a, b]$.
\end{proof} 

\begin{exercise}{3}
If $Y$ is the eigenspace corresponding to an eigenvalue $\lambda$ of an operator $T$, what is the spectrum of $T|_Y$?
\end{exercise}
\begin{proof}
It is precisely $\lambda$, because $\lambda$ is the value that solves $(T|_Y x - \lambda I)x = 0$, for all $x \in Y$.
\end{proof} 

\begin{exercise}{4}
Let $T: \ell^2 \to \ell^2$ be defined by $y = Tx, x = (x_n), y = (y_n), y_n = \alpha_n x_n$, where $(\alpha_n)$ is dense in $[0, 1]$.
Find $\sigma_p(T)$ and $\sigma(T)$.
\end{exercise}
\begin{proof}
$\sigma_p(T)$ is composed of all points in $(\alpha_n)$.
To see this, choose $x = (x_n)$ with 0 everywhere and $1$ at the $i$'th position and choose $\lambda = \alpha_i$.
Then $T x = \alpha_i = \lambda x$, as required.

For the rest of the spectrum, note that $(\alpha_n)$ is dense in $[0, 1]$, and that $\sigma(T)$ must be closed, thus $\sigma(T) = \overline{(\alpha_n)} = [0, 1]$.
\end{proof} 

\begin{exercise}{5}
If $\lambda \in \sigma(T) \setminus \sigma_p(T)$ in exercise 4, show that $R_\lambda(T)$ is unbounded.
\end{exercise}
\begin{proof}
Let $\lambda \in \sigma(T) \setminus \sigma_p(T)$, then $R_\lambda(T)(x)_j = (T - \lambda I)^{-1}(x_j) = x_j/(\alpha_j - \lambda)$.
Since $(\alpha_n)$ is dense in $[0, 1]$ for all $\epsilon > 0$, there exists an $\alpha \in (\alpha_n)$ such that $\absoluteValue{\alpha - \lambda} < \epsilon$.
Let $K$ be a candidate bound for $R_\lambda(T)$, so that for all $x$ with $\norm{x} = 1$, it should hold that $\norm{Tx} < K$.
Thus if we choose $\alpha \in (\alpha_n)$ with $\absoluteValue{\alpha - \lambda}< 1/K$ and we choose $x$ to be 0 everywhere and 1 at $i$ where $\alpha = \alpha_i$, then $\norm{x} = 1$ and also $\norm{Tx} = 1/\absoluteValue{\alpha - \lambda} = K$, contradicting the bound.
\end{proof}
