\section{Spectral properties of bounded linear operators}


\begin{exercise}{1}
Let $X = C[0, 1]$ and define $T: X \to X$ by $Tf = gf$, where $g \in X$ is fixed.
Find $\sigma(T)$.
Note that $\sigma(T)$ is closed.
\end{exercise}
\begin{proof}
We want to find $\lambda$ such that $R_\lambda(T) = (T - \lambda I)^{-1}$ either doesn't exists, is not bounded or is not defined on a dense subset of $X$.
For a fixed $\lambda$, $R_\lambda(T) = f(t) / (g(t) - \lambda)$, so that $R_\lambda(T)$ does not exist if $\lambda \in \cR(g)$.
Furthermore, since $g \in C[0, 1]$, then $g([0,1])$ is compact, so that $\sigma(T)$ is closed.
\end{proof} 

\begin{exercise}{2}
Find a linear operator $T: C[0, 1] \to C[0, 1]$ whose spectrum is a given interval $[a, b]$.
\end{exercise}
\begin{proof}
Based on our previous solution, choose $g \in C([0,1])$ with $\cR(g) = [a, b]$.
\end{proof} 

\begin{exercise}{3}
If $Y$ is the eigenspace corresponding to an eigenvalue $\lambda$ of an operator $T$, what is the spectrum of $T|_Y$?
\end{exercise}
\begin{proof}
It is precisely $\lambda$, because $\lambda$ is the value that solves $(T|_Y x - \lambda I)x = 0$, for all $x \in Y$.
\end{proof} 

\begin{exercise}{4}
Let $T: \ell^2 \to \ell^2$ be defined by $y = Tx, x = (x_n), y = (y_n), y_n = \alpha_n x_n$, where $(\alpha_n)$ is dense in $[0, 1]$.
Find $\sigma_p(T)$ and $\sigma(T)$.
\end{exercise}
\begin{proof}
$\sigma_p(T)$ is composed of all points in $(\alpha_n)$.
To see this, choose $x = (x_n)$ with 0 everywhere and $1$ at the $i$'th position and choose $\lambda = \alpha_i$.
Then $T x = (0, \dots, 0, \alpha_i, 0, \dots) = \lambda x$, as required.

For the rest of the spectrum, note that $(\alpha_n)$ is dense in $[0, 1]$, and that $\sigma(T)$ must be closed, thus $\sigma(T) = \overline{(\alpha_n)} = [0, 1]$.
Furthermore, to see no other points are in $\sigma(T)$, note that by the spectral radius is given by $\absoluteValue{\lambda} \leq \norm{T}$ (Theorem 7.3-4), which in our case, for $x$ of norm 1, gives us $\norm{Tx} \leq \sup_i(\alpha_i) \leq 1$ (by choosing $x_i = 1$ with $\alpha_i$ arbitrarily close to 1).

We finish the proof by proving that no negative $\lambda$ are in the spectrum.
We do this by proving that the resolvent of $T$ for any $\lambda$ is bounded.
First, for a \emph{fixed} $\lambda < 0$, we have that $R_\lambda(T)(x)_i = x_i/(\alpha_i - \lambda)$.
Since $\alpha_i \geq 0$, then we have that $1/(\alpha_i - \lambda) < 1/\absoluteValue{\lambda} = K$.
Thus, for any $x$ of norm 1, we have that $\norm{R_\lambda(T)(x)} \leq K$, meaning the resolvent is bounded, completing the proof.
\end{proof} 

\begin{exercise}{5}
If $\lambda \in \sigma(T) \setminus \sigma_p(T)$ in exercise 4, show that $R_\lambda(T)$ is unbounded.
\end{exercise}
\begin{proof}
Let $\lambda \in \sigma(T) \setminus \sigma_p(T)$, then $R_\lambda(T)(x)_j = (T - \lambda I)^{-1}(x_j) = x_j/(\alpha_j - \lambda)$.
Since $(\alpha_n)$ is dense in $[0, 1]$ for all $\epsilon > 0$, there exists an $\alpha \in (\alpha_n)$ such that $\absoluteValue{\alpha - \lambda} < \epsilon$.
Let $K$ be a candidate bound for $R_\lambda(T)$, so that for all $x$ with $\norm{x} = 1$, it should hold that $\norm{Tx} < K$.
Thus if we choose $\alpha \in (\alpha_n)$ with $\absoluteValue{\alpha - \lambda}< 1/K$ and we choose $x$ to be 0 everywhere and 1 at $i$ where $\alpha = \alpha_i$, then $\norm{x} = 1$ and also $\norm{Tx} = 1/\absoluteValue{\alpha - \lambda} = K$, contradicting the bound.
\end{proof} 

\begin{exercise}{6}
Extending exercise 4, find a linear operator $T: \ell^2 \to \ell^2$ whose eigenvalues are dense in a given compact set $K \subseteq \C$ and $\sigma(T) = K$.
\end{exercise}
\begin{proof}
Choose $T$ as in exercise 4 with $(\alpha_n)$ dense in $K$.
We know this is possible because compact sets are separable (Carothers exercise 8.7).
\end{proof} 

\begin{exercise}{7}
Let $T \in B(X, X)$.
Show that $\norm{R_\lambda(T)} \to 0$ as $\lambda \to \infty$.
\end{exercise}
\begin{proof}
From equation 9 in the proof of Theorem 7.3-4, we have that $R_\lambda (T) = -(1 / \lambda) \sum_n^\infty (T / \lambda)^n$.
Taking the operator norm on both sides of the equality, we obtain $\norm{R_\lambda (T)} = (1 / \lambda) \norm{\sum_n^\infty (T / \lambda)^n}$.
For a large enough $\lambda$, we get that $\norm{T}/\lambda < 1$, so that on the right we will have a convergent geometric series (which also decreases in value as $\lambda \to \infty$).
Thus, $\norm{R_\lambda (T)} = (1 / \lambda) x_n$ for a decreasing bounded by 0 sequence, so that $\norm{R_\lambda (T)} \to 0$, completing the proof.
\end{proof} 

\begin{exercise}{8}
Let $X = C[0, \pi]$ and define $T: \cD(T) \to X$ by $f \mapsto f''$, where $\cD(T) = \set{f \in X: f', f'' \in X, f(0) = f(\pi) = 0}$.
Show that $\sigma(T)$ is not compact.
\end{exercise}
\begin{proof}
Consider $f(x) = \sin (n x)$, with $f''(x) = -n^2 \sin (n x)$, both of with belong to $\cD(T)$, as long as $n \in \Z$.
Thus, $n \in \Z$ gives us the eigenvalue $\lambda = -n^2$, since $T f = f'' = -n^2 \sin (n x) = -n^2 f = \lambda f$, meaning that $\sigma(T)$ is unbounded, proving that $\sigma(T)$ is not compact.
\end{proof} 

\begin{exercise}{9}
Let $T: \ell^\infty \to \ell^\infty$ be defined by $x \mapsto (x_2, x_3, \dots)$, where $x$ is given by $x = (x_1, x_2, \dots)$.
\begin{enumerate}
    \item If $\absoluteValue{\lambda} > 1$, show that $\lambda \in \rho(T)$.
    \item If $\absoluteValue{\lambda} \leq 1$, show that $\lambda$ is an eigenvalue and find the eigenspace $Y$.
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item In exercise 3.9.10 we concluded that $\norm{T} = 1$.
    Furthermore, the spectrum of $T$ lies in the disk given by $\absoluteValue{\lambda} \leq \norm{T}$ (Theorem 7.3-4), so that if $\absoluteValue{\lambda} > 1$, $\lambda \in \rho(T)$, as required.
    \item Let $x \in \ell^\infty$ and $\absoluteValue{\lambda} \leq 1$ be an eigenvalue.
    Note that $T x = \lambda x$ implies that $\lambda x = (\lambda x_1, \lambda x_2, \dots) = (x_2, x_3, \dots)$, meaning that $x_n = \lambda x_{n-1} = \lambda^2 x_{n-2} = \dots$, which in turn implies that $(x_n)$ is monotonically decreasing, so that $x \in \ell^\infty$.
    The recurrence relation gives us a clue of what the eigenspace for each $\lambda$ is, namely, the sequences in $\ell^\infty$ that decrease at a rate of $\lambda$.
\end{enumerate}
\end{proof} 

\begin{exercise}{10}
Let $T: \ell^p \to \ell^p$ be defined by $x \mapsto (x_2, x_3, \dots)$, where $x$ is given by $x = (x_1, x_2, \dots)$ and $1 \leq p < \infty$.
If $\absoluteValue{\lambda} = 1$, is $\lambda$ an eigenvalue of $T$ (as in exercise 9)?
\end{exercise}
\begin{proof}
It isn't.
Recall that for an element to be in $\ell^p$, it must hold that $\sum \absoluteValue{x_n}^p < \infty$.
Thus $T x = \lambda x = x$ only holds if $x$ is constant, in which case $\sum \absoluteValue{x_n}^p = \infty$, or $x = 0$, so that $\absoluteValue{\lambda} = 1$ is not an eigenvalue.
\end{proof}
