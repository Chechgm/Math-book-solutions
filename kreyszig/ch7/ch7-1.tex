\section{Spectral theory in finite dimensional normed spaces}


\begin{exercise}{1}
Find the eigenvalues and eigenvectors of the following matrices, where $a$ and $b$ are real and $b \neq 0$.
\begin{align*}
    &A = 
    \begin{bmatrix}
        1 & 2\\
        -8 & 11
    \end{bmatrix},\\
    &B =
    \begin{bmatrix}
        a & b\\
        -b & a
    \end{bmatrix}.
\end{align*}
\end{exercise}
\begin{proof}
We will only find the eigenvalues and corresponding eigenvectors for $B$.
We have that $\det(B - \lambda I) = (a-\lambda)^2 + b^2 = \lambda^2 - 2a\lambda + a^2 + b^2$. 
Using the quadratic formula, $(-b' \pm \sqrt{b'^2 - 4a'c'})/2a'$ (with $a' = 1, b' = 2a$ and $c' = a^2 + b^2$), we get $\lambda = a \pm ib$.

Replacing $\lambda = a + ib$ in $B - \lambda I$, we get $-ib x_1 + b x_2 = 0$ (by taking the first row, without loss of generality).
Thus, the eigenvector associated with $\lambda = a + ib$ is $\begin{bmatrix} 1 & i \end{bmatrix}'$.
Doing the same process for $\lambda = a - bi$, we find that its eigenvector is $\begin{bmatrix} 1 & -i \end{bmatrix}'$, completing the exercise.
\end{proof} 

\begin{exercise}{2 (Hermitian matrix)}
Show that the eigenvalues of a Hermitian matrix $A = (a_{jk})$ are real.
\end{exercise}
\begin{proof}
Let $A$ be Hermitian, so that $\bar{A}^T = A$.
Let $\lambda$ be an eigenvalue of $A$ with corresponding eigenvector $\overline{x}$;
that is, $A x = \lambda x$.
We have $\overline{x}^T A x = \lambda \overline{x}^T x$ which is the same as $\lambda = \overline{x}^T A x / \overline{x}^T x$.
We will prove this scalar is real, since the denominator is the norm of $x$, it is real.
To see the numerator is real, let $B = \overline{x}^T A x$, and notice that 
\begin{align*}
    \overline{B} 
    = \overline{B}^T 
    = (\overline{\overline{x}^T A x})^T 
    = (x^T \overline{A} \overline{x})^T
    = \overline{x}^T \overline{A}^T x
    = \overline{x}^T A x
    = B
\end{align*}
where the first equality follows because $B$ is a scalar and the second to last equality follows because $A$ is Hermitian.
Since $\overline{B} = B $, $B$ must be real and thus the eigenvalues of $A$ must be real.
\end{proof} 

\begin{exercise}{3 (Skew-Hermitian matrix)}
Show that the eigenvalues of a skew-hermitian matrix $A = (a_{jk})$ are pure imaginary or zero.
\end{exercise}
\begin{proof}
Using the same strategy as in the previous exercise (we just need to include a negative in the second to last equality), we can arrive to the conclusion that $\overline{B} = -B$, which implies that either $B = 0$, or $B$ is pure imaginary, as required.
\end{proof} 

\begin{exercise}{4 (Unitary matrix)}
Show that the eigenvalues of a unitary matrix have absolute value 1.
\end{exercise}
\begin{proof}
Let $\lambda$ be an eigenvalue of the unitary matrix $A$ ($\overline{A}^T = A^{-1}$, with corresponding eigenvector $x$ so that $A x = \lambda x$.
This implies that $\overline{x^T} \overline{A}^T = (\overline{A x})^T = \overline{\lambda x^T}$.
Multiplying the transformed equality by the usual one, we obtain $\overline{x^T} \overline{A}^T A x = \overline{x^T} A^{-1} A x = \overline{x^T} x$ on the right hand side, and $\overline{\lambda} \lambda \overline{x}^T x = \absoluteValue{\lambda} \overline{x}^T x$, on the left hand side.
Thus $\absoluteValue{\lambda} = 1$, completing the proof.
\end{proof} 

\begin{exercise}{5}
Let $X$ be a finite dimensional inner product space and $T: X \to X$ a linear operator.
If $T$ is self-adjoint, show that its spectrum is real.
If $T$ is unitary, show that its eigenvalues have absolute value 1.
\end{exercise}
\begin{proof}
This is the content of exercises 2 and 4, since the matrix that represents a self-adjoint operator in a finite dimensional normed space is precisely a Hermitian matrix, and likewise for unitary matrices.
\end{proof} 

\begin{exercise}{6 (Trace)}
Let $\lambda_1, \dots, \lambda_n$ be the $n$ eigenvalues of an $n$-rowed square matrix $A = (a_{jk})$, where some or all of the $\lambda_j$'s may be equal.
Show that the product of the eigenvalues equals $\det A$ and their sum equals the trace of $A$, that is, the sum of the elements of the principal diagonal: $\operatorname{trace} A = a_{11} + \dots + a_{nn}$.
\end{exercise}
\begin{proof}
We know that every operator over $\C$ has an upper triangular matrix (5.27 in Axler LADR, 3rd edition) and that those elements of the diagonal correspond to the eigenvalues of the operator (5.32 in LADR).
Since the determinant of an upper diagonal matrix is computed by multiplying the elements of the diagonal and the determinant does not depend on the basis of the matrix (10.41 LADR), we get the desired result.
Likewise, the trace is independent on the basis of the matrix (10.15 LADR), then the sum of the eigenvalues (the diagonal) is the trace.
\end{proof} 

\begin{exercise}{7 (Inverse)}
Show that the inverse $A^{-1}$ of a square matrix $A$ exists if and only if all the eigenvalues $\lambda_1, \dots, \lambda_n$ of $A$ are different from zero.
If $A^{-1}$ exists, show that it has the eigenvalues $1/\lambda_1, \dots, 1/\lambda_n$.
\end{exercise}
\begin{proof}
($\Rightarrow$)
Suppose by contrapositive that $\lambda = 0$, then $A x = \lambda x = 0$ for some $x \neq 0$, meaning that $A$ is no injective (because $A$ maps 0 to 0).
Since a matrix represents an operator in a finite dimensional space, injectivity is required for $A$ to be invertible, and thus $A$ is not invertible, as required.

($\Leftarrow$)
Suppose by contrapositive that $A$ is not invertible, so that $\det A = 0$, thus $\det(A - 0I) =0$, and $\lambda = 0$ is an eigenvalue of $A$.
\end{proof} 

\begin{exercise}{9}
If a square matrix $A = (a_{jk})$ has eigenvalues $\lambda_j$, $j = 1, \dots, n$ show that $kA$ has the eigenvalues $k \lambda_j$ and $A^m$ ($m \in \N$) has the eigenvalues $\lambda_j^m$.
\end{exercise}
\begin{proof}
Let $\lambda$ be an eigenvalue of $A$ with corresponding eigenvector $x$, so that $A x = \lambda x$.
We have $kA x = k\lambda x$, so that $k\lambda$ is an eigenvalue of $kA$.
Furthermore, we have that $A^2 x = A\lambda x = \lambda^2 x$, so that by an inductive argument we get that $A^n x = \lambda^n x$.
\end{proof} 

\begin{exercise}{10}
If $A$ is a square matrix with eigenvalues $\lambda_1, \dots, \lambda_n$ and $p$ is any polynomial, show that the matrix $p(A)$ has the eigenvalues $p(\lambda_j)$, $j = 1, \dots, n$.
\end{exercise}
\begin{proof}
This is a Corollary to the previous result.
Let $p$ be a polynomial so that $p(A)$ is a scalar combination of powers of $A$.
That is, $p(A) = \sum_{j=0}^N k_j A^j$.
Using the result of exercise 9 on each of the summands, we get $p(A) x = \sum_{j=0}^N k_j A^j = \sum_{j=0}^N k_j \lambda^j x = p(\lambda) x$, completing the proof.
\end{proof} 

\begin{exercise}{11}
If $x_j$ is an eigenvector of an $n$-rowed square matrix $A$ corresponding to an eigenvalue $\lambda_j$ and $C$ is a nonsingular $n$-rowed square matrix, show that $\lambda_j$ is an eigenvalue of $A' = C^{-1} A C$ and a corresponding eigenvector is $y_j = C^{-1}x_j$.
\end{exercise}
\begin{proof}
The proofs that the eigenvalues are the same is as in 7.1-3.
To see that $y = C^{-1} x$, note that $C^{-1} A C = C^{-1} A x = C^{-1} \lambda x = \lambda y$, as required.
\end{proof} 

\begin{exercise}{12}
Illustrate with a simple example that an $n$-rowed square matrix may not have eigenvectors which constitute a basis for $\R^n$ (or $\C^n$).
For instance, consider
\begin{align*}
    A =
    \begin{bmatrix}
        1 & 1\\
        0 & 1
    \end{bmatrix}.
\end{align*}
\end{exercise}
\begin{proof}
We have $\det (A - \lambda I) = (1 - \lambda)^2$, so that $\lambda = 1$.
Replacing $\lambda = 1$ in $A - \lambda I$, we get
\begin{align*}
    \begin{bmatrix}
        0 & 1\\
        0 & 0
    \end{bmatrix}.
\end{align*}
By looking at the null space of $A - \lambda I$, we get the following equations $0x_1 + x_2 = 0$ and $0x_1 + 0x_2 = 0$, giving us $\begin{bmatrix} 1 & 0\end{bmatrix}'$ and $\begin{bmatrix} 0 & 0 \end{bmatrix}'$.
These two do not span $\R^2$.
\end{proof}
